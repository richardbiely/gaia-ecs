59 void exec(World& world) {
60 [[maybe_unused]] auto& queryInfo = query.fetch();
61
62 #if GAIA_PROFILER_CPU
63 const auto name = entity_name(*queryInfo.world(), entity);
64 const char* pScopeName = !name.empty() ? name.data() : sc_system_query_func_str;
65 GAIA_PROF_SCOPE2(pScopeName);
66 #endif
67
68 auto* pRuntime = world.systems().data_try(entity);
69 GAIA_ASSERT(pRuntime != nullptr);
70 if (pRuntime == nullptr || !pRuntime->on_each_func)
71 return;
72
73 static_cast<void>(pRuntime->on_each_func(query, execType, SystemRuntimeData::RunMode::Immediate));
74 }

83 GAIA_NODISCARD mt::JobHandle job_handle(World& world) {
84 if (jobHandle == (mt::JobHandle)mt::JobNull_t{}) {
85 auto& tp = mt::ThreadPool::get();
86 auto func = [this, &world]() {
87 exec(world);
88 };
89 static_assert(mt::JobFunc::stores_inline<decltype(func)>(), "System jobs must not allocate");
90
91 mt::Job syncJob;
92 syncJob.func = GAIA_MOV(func);
93 syncJob.flags = mt::JobCreationFlags::ManualDelete;
94 jobHandle = tp.add(GAIA_MOV(syncJob));
95 }
96 return jobHandle;
97 }

114 return pRuntime->on_each_func(query, execType, SystemRuntimeData::RunMode::DeferredJob);

625 template <typename Func, std::enable_if_t<detail::is_query_iter_callback_v<Func>, int> = 0>
626 SystemBuilder& on_each(Func func) {
627 validate();
628
629 auto& runtime = runtime_data();
630 runtime.on_each_func = [func](Query& query, QueryExecType execType, SystemRuntimeData::RunMode mode) mutable {
631 if (mode == SystemRuntimeData::RunMode::DeferredJob)
632 return query.job(func, execType);
633
634 query.each_runtime_erased(
635 execType, static_cast<void*>(&func), &detail::QueryImpl::template invoke_runtime_iter<Func, Iter>,
636 Constraints::EnabledOnly);
637 return SchedJob{};
638 };
639
640 return (SystemBuilder&)*this;
641 }
