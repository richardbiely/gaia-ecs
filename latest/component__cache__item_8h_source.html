50 GAIA_NODISCARD SymbolId intern(util::str_view value) {
51 if (value.empty())
52 return {};
53
54 const Key lookup(value.data(), value.size(), 0);
55 const auto it = m_ids.find(lookup);
56 if (it != m_ids.end())
57 return it->second;
58
59 auto* data = mem::AllocHelper::alloc<char>(value.size() + 1);
60 memcpy(data, value.data(), value.size());
61 data[value.size()] = 0;
62 const Key owned(data, value.size(), 1);
63 const SymbolId id{(uint32_t)m_values.size()};
64 m_values.push_back(owned);
65 m_ids.emplace(owned, id);
66 return id;
67 }
68
72 GAIA_NODISCARD util::str_view view(SymbolId id) const noexcept {
73 if (!id.valid() || id.value >= m_values.size())
74 return {};
75 const auto& value = m_values[id.value];
76 return {value.str(), value.len()};
77 }
78
80 void clear() {
81 m_ids.clear();
82 for (const auto& value: m_values) {
83 if (value.str() != nullptr && value.owned())
84 mem::AllocHelper::free((void*)value.str());
85 }
86 m_values.clear();
87 }

245 auto* pD = mem::data_view_policy_soa_erased::set(pDst, comp.alig(), fieldSizes, i, idxDst, capDst);
246 const auto* pS = mem::data_view_policy_soa_erased::get(pSrc, comp.alig(), fieldSizes, i, idxSrc, capSrc);
247 memmove(pD, pS, soaSizes[i]);
248 }

700 template <typename T>
701 GAIA_NODISCARD static uint32_t init_type_name(char (&nameTmp)[MaxNameLength]) {
702 auto ct_name = detail::ComponentDesc<T>::name();
703
704 // Allocate enough memory for the name string + the null-terminating character (
705 // the compile time string returned by ComponentDesc<T>::name is not null-terminated).
706 // Different compilers will give a bit different strings, e.g.:
707 // Clang/GCC: gaia::ecs::uni<Position>
708 // MSVC : gaia::ecs::uni<struct Position>
709 // Therefore, we first copy the compile-time string and then tweak it so it is
710 // the same on all supported compilers.
711 auto nameTmpLen = (uint32_t)ct_name.size();
712 GAIA_ASSERT(nameTmpLen < MaxNameLength);
713 memcpy((void*)nameTmp, (const void*)ct_name.data(), nameTmpLen);
714 nameTmp[ct_name.size()] = 0;
715
716 auto strip_prefix = [&](const char* prefix, uint32_t prefixLen) {
717 if (nameTmpLen <= prefixLen || strncmp(nameTmp, prefix, prefixLen) != 0)
718 return;
719
720 memmove(nameTmp, nameTmp + prefixLen, nameTmpLen - prefixLen + 1);
721 nameTmpLen -= prefixLen;
722 };
723
724 strip_prefix("const ", 6);
725
726 const uint32_t NSubstrings = 3;
727 const char* to_remove[NSubstrings] = {"class ", "struct ", "enum "};
728 const uint32_t to_remove_len[NSubstrings] = {6, 7, 5};
729 GAIA_FOR(NSubstrings) {
730 strip_prefix(to_remove[i], to_remove_len[i]);
731 }
732
733 while (nameTmpLen > 0) {
734 const auto ch = nameTmp[nameTmpLen - 1];
735 if (ch != ' ' && ch != '&' && ch != '*')
736 break;
737
738 nameTmp[--nameTmpLen] = 0;
739 }
740
741 if (nameTmpLen > 6 && strncmp(nameTmp + nameTmpLen - 6, " const", 6) == 0) {
742 nameTmpLen -= 6;
743 nameTmp[nameTmpLen] = 0;
744 }
745
746 // Normalization template names by removing keywords when they appear as template argument
747 // prefixes instead of as part of a longer identifier.
748 GAIA_FOR(NSubstrings) {
749 const auto* str = to_remove[i];
750 const auto len = to_remove_len[i];
751
752 auto* pos = nameTmp;
753 while ((pos = strstr(pos, str)) != nullptr) {
754 const bool isBoundary = pos == nameTmp || pos[-1] == '<' || pos[-1] == ',' || pos[-1] == ' ';
755 if (!isBoundary) {
756 ++pos;
757 continue;
758 }
759
760 const auto tailMaxLen = (size_t)(MaxNameLength - (uint32_t)(pos + len - nameTmp));
761 memmove(pos, pos + len, GAIA_STRLEN(pos + len, tailMaxLen) + 1);
762 nameTmpLen -= len;
763 }
764 }
765
766 return nameTmpLen;
767 }

809 template <typename T>
810 GAIA_NODISCARD static ComponentCacheItem*
811 create_typed(Entity entity, SymbolTable& symbols, const RuntimeTypeDesc* pRuntimeType) {
812 static_assert(core::is_raw_v<T>);
813
814 constexpr auto componentSize = detail::ComponentDesc<T>::size();
815 static_assert(
816 componentSize < Component::MaxComponentSizeInBytes,
817 "Trying to register a component larger than the maximum allowed component size! In the future this "
818 "restriction won't apply to components not stored inside archetype chunks.");
819
820 char nameTmp[MaxNameLength];
821 const auto nameTmpLen = init_type_name<T>(nameTmp);
822
