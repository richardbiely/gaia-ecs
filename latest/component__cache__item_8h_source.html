50 GAIA_NODISCARD SymbolId intern(util::str_view value) {
51 if (value.empty())
52 return {};
53
54 const Key lookup(value.data(), value.size(), 0);
55 const auto it = m_ids.find(lookup);
56 if (it != m_ids.end())
57 return it->second;
58
59 auto* data = mem::AllocHelper::alloc<char>(value.size() + 1);
60 memcpy(data, value.data(), value.size());
61 data[value.size()] = 0;
62 const Key owned(data, value.size(), 1);
63 const SymbolId id{(uint32_t)m_values.size()};
64 m_values.push_back(owned);
65 m_ids.emplace(owned, id);
66 return id;
67 }
68
72 GAIA_NODISCARD util::str_view view(SymbolId id) const noexcept {
73 if (!id.valid() || id.value >= m_values.size())
74 return {};
75 const auto& value = m_values[id.value];
76 return {value.str(), value.len()};
77 }
78
80 void clear() {
81 m_ids.clear();
82 for (const auto& value: m_values) {
83 if (value.str() != nullptr && value.owned())
84 mem::AllocHelper::free((void*)value.str());
85 }
86 m_values.clear();
87 }

692 template <typename T>
693 GAIA_NODISCARD static uint32_t init_type_name(char (&nameTmp)[MaxNameLength]) {
694 auto ct_name = detail::ComponentDesc<T>::name();
695
696 // Allocate enough memory for the name string + the null-terminating character (
697 // the compile time string returned by ComponentDesc<T>::name is not null-terminated).
698 // Different compilers will give a bit different strings, e.g.:
699 // Clang/GCC: gaia::ecs::uni<Position>
700 // MSVC : gaia::ecs::uni<struct Position>
701 // Therefore, we first copy the compile-time string and then tweak it so it is
702 // the same on all supported compilers.
703 auto nameTmpLen = (uint32_t)ct_name.size();
704 GAIA_ASSERT(nameTmpLen < MaxNameLength);
705 memcpy((void*)nameTmp, (const void*)ct_name.data(), nameTmpLen);
706 nameTmp[ct_name.size()] = 0;
707
708 auto strip_prefix = [&](const char* prefix, uint32_t prefixLen) {
709 if (nameTmpLen <= prefixLen || strncmp(nameTmp, prefix, prefixLen) != 0)
710 return;
711
712 memmove(nameTmp, nameTmp + prefixLen, nameTmpLen - prefixLen + 1);
713 nameTmpLen -= prefixLen;
714 };
715
716 strip_prefix("const ", 6);
717
718 const uint32_t NSubstrings = 3;
719 const char* to_remove[NSubstrings] = {"class ", "struct ", "enum "};
720 const uint32_t to_remove_len[NSubstrings] = {6, 7, 5};
721 GAIA_FOR(NSubstrings) {
722 strip_prefix(to_remove[i], to_remove_len[i]);
723 }
724
725 while (nameTmpLen > 0) {
726 const auto ch = nameTmp[nameTmpLen - 1];
727 if (ch != ' ' && ch != '&' && ch != '*')
728 break;
729
730 nameTmp[--nameTmpLen] = 0;
731 }
732
733 if (nameTmpLen > 6 && strncmp(nameTmp + nameTmpLen - 6, " const", 6) == 0) {
734 nameTmpLen -= 6;
735 nameTmp[nameTmpLen] = 0;
736 }
737
738 // Normalization template names by removing keywords when they appear as template argument
739 // prefixes instead of as part of a longer identifier.
740 GAIA_FOR(NSubstrings) {
741 const auto* str = to_remove[i];
742 const auto len = to_remove_len[i];
743
744 auto* pos = nameTmp;
745 while ((pos = strstr(pos, str)) != nullptr) {
746 const bool isBoundary = pos == nameTmp || pos[-1] == '<' || pos[-1] == ',' || pos[-1] == ' ';
747 if (!isBoundary) {
748 ++pos;
749 continue;
750 }
751
752 const auto tailMaxLen = (size_t)(MaxNameLength - (uint32_t)(pos + len - nameTmp));
753 memmove(pos, pos + len, GAIA_STRLEN(pos + len, tailMaxLen) + 1);
754 nameTmpLen -= len;
755 }
756 }
757
758 return nameTmpLen;
759 }

800 GAIA_NODISCARD static ComponentCacheItem*
801 create_typed(Entity entity, SymbolTable& symbols, const RuntimeTypeDesc* pRuntimeType) {
802 static_assert(core::is_raw_v<T>);
803
804 constexpr auto componentSize = detail::ComponentDesc<T>::size();
805 static_assert(
806 componentSize < Component::MaxComponentSizeInBytes,
807 "Trying to register a component larger than the maximum allowed component size! In the future this "
808 "restriction won't apply to components not stored inside archetype chunks.");
809
810 char nameTmp[MaxNameLength];
811 const auto nameTmpLen = init_type_name<T>(nameTmp);
812
813 uint8_t soaSizes[meta::StructToTupleMaxTypes]{};
814 RuntimeFieldInit fields[meta::StructToTupleMaxTypes]{};
815 auto desc = detail::ComponentDesc<T>::make(
816 util::str_view(nameTmp, nameTmpLen), std::span<uint8_t, meta::StructToTupleMaxTypes>{soaSizes});
817 if (pRuntimeType != nullptr) {
818 desc.runtimeType = *pRuntimeType;
