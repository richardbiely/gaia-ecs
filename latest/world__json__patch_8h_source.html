15 inline bool runtime_patch_decode_token(ser::json_str_view encoded, util::str& token) {
16 token.clear();
17
18 GAIA_FOR(encoded.size()) {
19 const auto ch = encoded.data()[i];
20 if (ch != '~') {
21 token.append(ch);
22 continue;
23 }
24
25 if (i + 1 >= encoded.size())
26 return false;
27
28 const auto escaped = encoded.data()[++i];
29 if (escaped == '0')
30 token.append('~');
31 else if (escaped == '1')
32 token.append('/');
33 else
34 return false;
35 }
36
37 return true;
38 }

44 inline bool runtime_patch_parse_index(util::str_view token, uint32_t& index) {
45 if (token.empty())
46 return false;
47
48 uint64_t value = 0;
49 GAIA_FOR(token.size()) {
50 const auto ch = token.data()[i];
51 if (ch < '0' || ch > '9')
52 return false;
53 value = value * 10 + (uint32_t)(ch - '0');
54 if (value > UINT32_MAX)
55 return false;
56 }
57
58 index = (uint32_t)value;
59 return true;
60 }

109 inline bool World::patch_comp_json(
110 Entity entity, Entity component, ser::json_str_view pointer, ser::json_str_view value,
111 ser::JsonDiagnostics& diagnostics, const ser::RuntimeJsonPolicy& policy, uint64_t expectedRuntimeSchemaHash) {
112 auto error = [&](ser::JsonDiagReason reason, const char* message) {
113 diagnostics.add(ser::JsonDiagSeverity::Error, reason, pointer, message);
114 return false;
115 };
116
117 if (expectedRuntimeSchemaHash != 0 && expectedRuntimeSchemaHash != runtime_schema_hash())
118 return error(ser::JsonDiagReason::StaleSchema, "Runtime schema hash does not match the current manifest.");
119
120 const auto* pRoot = component.pair() ? m_compCache.find_pair_payload(component) : m_compCache.find(component);
121 if (pRoot == nullptr)
122 return error(ser::JsonDiagReason::UnknownComponent, "Component patch target is not registered.");
123
124 auto cursor = cursor_mut(entity, component);
125 if (!cursor.valid() || cursor.size() == 0)
126 return error(ser::JsonDiagReason::MissingComponentStorage, "Component patch payload is unavailable.");
127 if (!pointer.empty() && pointer.data()[0] != '/')
128 return error(ser::JsonDiagReason::InvalidPatchPath, "Component patch path must be an RFC 6901 JSON Pointer.");
129 if (!pointer.empty() && (pointer.size() == 1 || pointer.data()[pointer.size() - 1] == '/'))
130 return error(ser::JsonDiagReason::InvalidPatchPath, "Component patch path contains an empty token.");
131
132 const ComponentCacheItem* pCurrent = pRoot;
133 const RuntimeFieldDesc* pSelectedField = nullptr;
134 uint32_t pos = pointer.empty() ? pointer.size() : 1;
135 while (pos < pointer.size()) {
136 uint32_t end = pos;
137 while (end < pointer.size() && pointer.data()[end] != '/')
138 ++end;
139
140 util::str token;
141 if (!detail::runtime_patch_decode_token(ser::json_str_view(pointer.data() + pos, end - pos), token) ||
142 token.empty())
143 return error(ser::JsonDiagReason::InvalidPatchPath, "Component patch path contains an invalid token.");
144
145 const ComponentCacheItem* pFieldOwner = pCurrent;
146 if (pFieldOwner != nullptr && pFieldOwner->typeKind == RuntimeTypeKind::Opaque)
147 pFieldOwner = m_compCache.find(pFieldOwner->opaque_as_type());
148
149 if (pFieldOwner != nullptr && pFieldOwner->typeKind == RuntimeTypeKind::Struct) {
150 const auto* pField = pFieldOwner->field(util::str_view(token.data(), token.size()));
151 if (pField == nullptr)
152 return error(ser::JsonDiagReason::InvalidPatchPath, "Component patch field does not exist.");
153 if ((pField->flags & RuntimeFieldFlag_ReadOnly) != 0)
154 return error(ser::JsonDiagReason::ReadOnlyField, "Component patch field is read-only.");
155 if ((pField->flags & RuntimeFieldFlag_Hidden) != 0)
156 return error(ser::JsonDiagReason::HiddenField, "Component patch field is hidden.");
157 if (!cursor.field(util::str_view(token.data(), token.size())))
158 return error(ser::JsonDiagReason::InvalidPatchPath, "Component patch field cannot be traversed.");
159
160 pSelectedField = pField;
161 pCurrent = m_compCache.find(pField->type);
162 } else {
163 uint32_t index = 0;
164 if (!detail::runtime_patch_parse_index(util::str_view(token.data(), token.size()), index) ||
165 !cursor.elem(index))
166 return error(ser::JsonDiagReason::InvalidPatchPath, "Component patch sequence index is invalid.");
167
168 pCurrent = m_compCache.find(cursor.type());
169 }
170
171 pos = end + 1;
172 }
173
174 const auto count = cursor.count();
175 const bool charBuffer = pSelectedField != nullptr && cursor.type() == Char8 && count.ok() && count.value > 1;
176
177 if (pCurrent == nullptr || !detail::runtime_json_leaf_editable(*pCurrent) ||
178 (count.ok() && count.value > 1 && !charBuffer))
179 return error(
180 ser::JsonDiagReason::UnsupportedPatchValue, "Component patch endpoint must be a supported reflected leaf.");
181
182 // Work on a copy so a failed patch does not change the component.
183 cnt::darray<uint8_t> original;
184 original.resize(cursor.size());
185 if (!cursor.get_raw(original.data(), (uint32_t)original.size()))
186 return error(ser::JsonDiagReason::UnsupportedPatchValue, "Component patch endpoint cannot be read.");
187
188 cnt::darray<uint8_t> bytes;
189 bytes.resize(original.size());
190 memcpy(bytes.data(), original.data(), original.size());
191
192 ser::ser_json reader(value.data(), value.size());
193 bool valueOk = true;
194 detail::RuntimeJsonReadContext ctx{&m_compCache, reader, diagnostics, policy, valueOk};
195
196 if (cursor.type() == Char8 && cursor.size() == sizeof(char)) {
197 // One Char8 value is written as a one-character JSON string.
198 ser::json_str_view text;
199 if (!reader.parse_string_view(text) || text.size() != 1)
200 return error(ser::JsonDiagReason::InvalidJson, "Component patch character value must contain one character.");
201
202 bytes[0] = (uint8_t)text.data()[0];
203 } else if (!detail::read_runtime_json_value(
204 ctx, pCurrent, cursor.type(), bytes.data(), (uint32_t)bytes.size(), pointer, 0)) {
205 return error(
206 ser::JsonDiagReason::InvalidJson, "Component patch value is not valid JSON for the selected field.");
207 }
208
209 reader.ws();
210 if (!reader.eof())
211 return error(ser::JsonDiagReason::InvalidJson, "Component patch value contains trailing JSON.");
212
213 if (!valueOk)
214 return error(
215 ser::JsonDiagReason::UnsupportedPatchValue,
216 "Component patch value is incompatible with the selected field.");
217
218 if (pSelectedField != nullptr &&
219 (pSelectedField->flags & (RuntimeFieldFlag_HasMinimum | RuntimeFieldFlag_HasMaximum)) != 0) {
220 double number = 0.0;
221 if (!detail::runtime_patch_numeric_value(pCurrent, cursor.type(), bytes.data(), cursor.size(), number))
222 return error(
223 ser::JsonDiagReason::UnsupportedPatchValue, "Component patch range applies to a non-numeric field.");
224
225 if (((pSelectedField->flags & RuntimeFieldFlag_HasMinimum) != 0 && number < pSelectedField->minimum) ||
226 ((pSelectedField->flags & RuntimeFieldFlag_HasMaximum) != 0 && number > pSelectedField->maximum))
227 return error(
228 ser::JsonDiagReason::RangeViolation, "Component patch value is outside the authored field range.");
229 }
230
231 if (!cursor.set_raw(bytes.data(), (uint32_t)bytes.size())) {
232 // The write may be incomplete, so put the original value back.
233 (void)cursor.set_raw(original.data(), (uint32_t)original.size(), false);
234 return error(
235 ser::JsonDiagReason::UnsupportedPatchValue, "Component patch could not commit the selected field.");
236 }
237
238 return true;
239 }
