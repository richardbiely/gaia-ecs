110 template <typename T, span_size_type Extent>
111 class span {
112 static_assert(
113 std::is_object<T>::value, "A span's T must be an object type (not a "
114 "reference type or void)");
115 static_assert(
116 detail::is_complete<T>::value, "A span's T must be a complete type (not a forward "
117 "declaration)");
118 static_assert(!std::is_abstract<T>::value, "A span's T cannot be an abstract class type");
119
120 using m_datatype = detail::span_storage<T, Extent>;
121
122 public:
124 using element_kind = T;
126 using value_type = typename std::remove_cv<T>::type;
128 using size_type = span_size_type;
130 using difference_type = span_difference_type;
132 using pointer = element_kind*;
134 using const_pointer = const element_kind*;
136 using reference = element_kind&;
138 using const_reference = const element_kind&;
139
141 using iterator = pointer;
143 using const_iterator = const_pointer;
145 using iterator_type = core::random_access_iterator_tag;
146
148 static constexpr size_type extent = Extent;
149
150 private:
151 m_datatype m_data{};
152
153 public:
156 template <span_size_type E = Extent, typename std::enable_if<(E == DynamicSpanExtent || E <= 0), int>::type = 0>
157 constexpr span() noexcept {}
158
161 template <
162 typename U,
163 typename std::enable_if<std::is_convertible<U (*)[], element_kind (*)[]>::value, int>::type = 0>
164 span(std::initializer_list<U>) = delete;
165
169 constexpr span(pointer ptr, size_type count): m_data(ptr, count) {
170 GAIA_ASSERT(extent == DynamicSpanExtent || extent == count);
171 }
172
176 constexpr span(pointer begin, pointer end): m_data(begin, end - begin) {
177 GAIA_ASSERT(extent == DynamicSpanExtent || ((uintptr_t)(end - begin) == (uintptr_t)extent));
178 }
179
184 template <
185 span_size_type N, span_size_type E = Extent,
186 typename std::enable_if<
187 (E == DynamicSpanExtent || N == E) &&
188 detail::is_container_element_kind_compatible<element_kind (&)[N], T>::value,
189 int>::type = 0>
190 constexpr span(element_kind (&arr)[N]) noexcept: m_data(arr, N) {}
191
196 template <
197 typename Container, span_size_type E = Extent,
198 typename std::enable_if<
199 E == DynamicSpanExtent && detail::is_container<Container>::value &&
200 detail::is_container_element_kind_compatible<Container&, T>::value,
201 int>::type = 0>
202 constexpr span(Container& cont): m_data(detail::data(cont), detail::size(cont)) {}
203
208 template <
209 typename Container, span_size_type E = Extent,
210 typename std::enable_if<
211 E == DynamicSpanExtent && detail::is_container<Container>::value &&
212 detail::is_container_element_kind_compatible<const Container&, T>::value,
213 int>::type = 0>
214 constexpr span(const Container& cont): m_data(detail::data(cont), detail::size(cont)) {}
215
216 constexpr span(const span&) noexcept = default;
217 constexpr span& operator=(const span&) noexcept = default;
218
223 template <
224 typename T2, span_size_type Extent2,
225 typename std::enable_if<
226 (Extent == DynamicSpanExtent || Extent2 == DynamicSpanExtent || Extent == Extent2) &&
227 std::is_convertible<T2 (*)[], T (*)[]>::value,
228 int>::type = 0>
229 constexpr span(const span<T2, Extent2>& other) noexcept: m_data(other.data(), other.size()) {}

231 ~span() noexcept = default;

347 using subspan_return_t = span<

354 template <span_size_type Offset, span_size_type Count = DynamicSpanExtent>
355 GAIA_NODISCARD constexpr subspan_return_t<Offset, Count> subspan() const {
356 GAIA_ASSERT(Offset <= size() && (Count == DynamicSpanExtent || Offset + Count <= size()));
357 return {m_data.beg + Offset, Count != DynamicSpanExtent ? Count : size() - Offset};
