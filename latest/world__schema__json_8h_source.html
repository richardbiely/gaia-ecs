55 inline bool
56 runtime_schema_type_editable(const ComponentCache& cache, const ComponentCacheItem& item, uint32_t depth = 0) {
57 if (depth >= RuntimeJsonMaxDepth)
58 return false;
59
60 auto referenced_type_editable = [&](Entity type) {
61 const auto* pType = cache.find(type);
62 return pType != nullptr && runtime_schema_type_editable(cache, *pType, depth + 1);
63 };
64
65 switch (item.typeKind) {
66 case RuntimeTypeKind::Primitive:
67 case RuntimeTypeKind::Enum:
68 case RuntimeTypeKind::Bitmask:
69 return runtime_json_leaf_editable(item);
70 case RuntimeTypeKind::Struct:
71 GAIA_FOR(item.field_count()) {
72 const auto& field = *item.field(i);
73 if ((field.flags & (RuntimeFieldFlag_ReadOnly | RuntimeFieldFlag_Hidden)) == 0 &&
74 referenced_type_editable(field.type))
75 return true;
76 }
77 return false;
78 case RuntimeTypeKind::Array:
79 return item.elementCount > 0 && referenced_type_editable(item.elementType);
80 case RuntimeTypeKind::Vector:
81 return item.sequenceAdapter != nullptr && item.sequenceAdapter->count != nullptr &&
82 item.sequenceAdapter->element != nullptr && referenced_type_editable(item.elementType);
83 case RuntimeTypeKind::Opaque: {
84 if (item.opaqueAdapter == nullptr || item.opaqueAdapter->project == nullptr)
85 return false;
86 const auto* pProjected = cache.find(item.opaqueAsType);
87 return pProjected != nullptr &&
88 (pProjected->typeKind == RuntimeTypeKind::Struct || pProjected->typeKind == RuntimeTypeKind::Array ||
89 pProjected->typeKind == RuntimeTypeKind::Vector) &&
90 runtime_schema_type_editable(cache, *pProjected, depth + 1);
91 }
92 case RuntimeTypeKind::None:
93 return false;
94 }
95 return false;
96 }

125 GAIA_NODISCARD inline bool
126 runtime_schema_semantic_path(const World& world, Entity semantic, util::str& out, uint64_t& stableHash) {
127 out.clear();
128 stableHash = 0;
129 if (semantic == EntityBad || semantic.pair())
130 return false;
131
132 cnt::darray_ext<util::str_view, 16> segments;
133 auto current = semantic;
134 while (current != EntityBad) {
135 if (!world.valid(current) || current.pair())
136 return false;
137 const auto name = world.name(current);
138 if (name.empty())
139 return false;
140 segments.push_back(name);
141 current = world.target(current, ChildOf);
142 }
143 if (segments.empty())
144 return false;
145
146 uint32_t size = (uint32_t)segments.size() - 1;
147 for (const auto segment: segments)
148 size += segment.size();
149 out.reserve(size);
150 stableHash = core::calculate_hash64("gaia.ecs.semantic.path.v1");
151 for (uint32_t i = (uint32_t)segments.size(); i > 0; --i) {
152 const auto segment = segments[i - 1];
153 if (!out.empty())
154 out.append('.');
155 out.append(segment);
156 stableHash = core::hash_combine(
157 stableHash, (uint64_t)segment.size(), core::calculate_hash64(segment.data(), segment.size()));
158 }
159 return true;
160 }

216 writer.key("hidden");

218 writer.key("multiline");
219 writer.value_bool((field.flags & RuntimeFieldFlag_Multiline) != 0);
220 const auto unit = item.field_unit(field);
221 if (!unit.empty()) {
222 writer.key("unit");
223 writer.value_string(unit.data(), unit.size());
224 }
225 if ((field.flags & RuntimeFieldFlag_HasMinimum) != 0) {
226 writer.key("minimum");

228 }
229 if ((field.flags & RuntimeFieldFlag_HasMaximum) != 0) {
230 writer.key("maximum");
231 writer.value_float(field.maximum);

234 writer.key("step");
235 writer.value_float(field.step);
236 }
237 RuntimeJsonFieldLayout layout{};

239 runtime_json_is_char8_type(layout.pType, layout.type);
240 if (field.jsonEncoding == RuntimeJsonEncoding::Utf8String || derivedUtf8) {
241 writer.key("encoding");

243 writer.key("capacity");

356 inline bool
357 World::write_runtime_schema_json(ser::ser_json& writer, const char* schemaHash, bool includeRuntimeEntities) const {
358 cnt::darray<const ComponentCacheItem*> items;
359 items.reserve(m_compCache.m_compByEntityId.size());
360 for (const auto& [entityId, pItem]: m_compCache.m_compByEntityId) {
361 (void)entityId;
362 items.push_back(pItem);
363 }
364 core::sort(items.begin(), items.end(), [](const ComponentCacheItem* a, const ComponentCacheItem* b) {
365 const auto nameOrder = strcmp(a->symbol_name().data(), b->symbol_name().data());
366 if (nameOrder != 0)
367 return nameOrder < 0;
368 const auto pathSize = a->path.size() < b->path.size() ? a->path.size() : b->path.size();
369 const auto pathOrder = pathSize != 0 ? memcmp(a->path.data(), b->path.data(), pathSize) : 0;
370 return pathOrder != 0 ? pathOrder < 0 : a->path.size() < b->path.size();
371 });
372 writer.clear();
373 writer.begin_object();
374 writer.key("format");
375 writer.value_string("gaia.ecs.schema");
376 writer.key("version");
377 writer.value_int(2);
378 if (schemaHash != nullptr) {
379 writer.key("hash");
380 writer.value_string(schemaHash);
381 }
382 writer.key("types");
383 writer.begin_array();
384 for (const auto* pItem: items)
385 detail::runtime_schema_write_item(writer, *this, m_compCache, *pItem, includeRuntimeEntities);
386 writer.end_array();
387 writer.end_object();
388 return true;
389 }
