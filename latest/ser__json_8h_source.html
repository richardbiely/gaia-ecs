205 GAIA_NODISCARD static bool parse_hex_quad(const char* str, uint32_t& value) {
206 value = 0;
207 GAIA_FOR(4) {
208 const char ch = str[i];
209 uint32_t digit = 0;
210 if (ch >= '0' && ch <= '9')
211 digit = (uint32_t)(ch - '0');
212 else if (ch >= 'a' && ch <= 'f')
213 digit = (uint32_t)(ch - 'a') + 10;
214 else if (ch >= 'A' && ch <= 'F')
215 digit = (uint32_t)(ch - 'A') + 10;
216 else
217 return false;
218 value = (value << 4U) | digit;
219 }
220 return true;
221 }

223 static void append_utf8(json_str& out, uint32_t codePoint) {
224 if (codePoint <= 0x7fU) {
225 out.append((char)codePoint);
226 } else if (codePoint <= 0x7ffU) {
227 out.append((char)(0xc0U | (codePoint >> 6U)));
228 out.append((char)(0x80U | (codePoint & 0x3fU)));
229 } else if (codePoint <= 0xffffU) {
230 out.append((char)(0xe0U | (codePoint >> 12U)));
231 out.append((char)(0x80U | ((codePoint >> 6U) & 0x3fU)));
232 out.append((char)(0x80U | (codePoint & 0x3fU)));
233 } else {
234 out.append((char)(0xf0U | (codePoint >> 18U)));
235 out.append((char)(0x80U | ((codePoint >> 12U) & 0x3fU)));
236 out.append((char)(0x80U | ((codePoint >> 6U) & 0x3fU)));
237 out.append((char)(0x80U | (codePoint & 0x3fU)));
238 }
239 }

481 bool parse_string_view(json_str_view& out, bool* fromScratch = nullptr) {
482 ws();
483 if (m_it == nullptr || m_end == nullptr || m_it >= m_end || *m_it != '"')
484 return false;
485
486 ++m_it;
487 const char* begin = m_it;
488 bool escaped = false;
489 m_parseScratch.clear();
490 while (m_it < m_end) {
491 const char ch = *m_it++;
492 if (ch == '"')
493 break;
494
495 if (ch == '\\') {
496 if (!escaped) {
497 escaped = true;
498 const auto prefixLen = (size_t)((m_it - 1) - begin);
499 if (prefixLen > 0)
500 m_parseScratch.append(begin, (uint32_t)prefixLen);
501 }
502
503 if (m_it >= m_end)
504 return false;
505 const char esc = *m_it++;
506 switch (esc) {
507 case '"':
508 case '\\':
509 case '/':
510 m_parseScratch.append(esc);
511 break;
512 case 'b':
513 m_parseScratch.append('\b');
514 break;
515 case 'f':
516 m_parseScratch.append('\f');
517 break;
518 case 'n':
519 m_parseScratch.append('\n');
520 break;
521 case 'r':
522 m_parseScratch.append('\r');
523 break;
524 case 't':
525 m_parseScratch.append('\t');
526 break;
527 case 'u': {
528 if ((uint32_t)(m_end - m_it) < 4)
529 return false;
530 uint32_t codePoint = 0;
531 if (!parse_hex_quad(m_it, codePoint))
532 return false;
533 m_it += 4;
534 if (codePoint >= 0xd800U && codePoint <= 0xdbffU) {
535 if ((uint32_t)(m_end - m_it) < 6 || m_it[0] != '\\' || m_it[1] != 'u')
536 return false;
537 uint32_t low = 0;
538 if (!parse_hex_quad(m_it + 2, low) || low < 0xdc00U || low > 0xdfffU)
539 return false;
540 m_it += 6;
541 codePoint = 0x10000U + ((codePoint - 0xd800U) << 10U) + (low - 0xdc00U);
542 } else if (codePoint >= 0xdc00U && codePoint <= 0xdfffU) {
543 return false;
544 }
545 append_utf8(m_parseScratch, codePoint);
546 break;
547 }
548 default:
549 return false;
550 }
551 } else if (escaped)
552 m_parseScratch.append(ch);
553 }
554
555 if (m_it <= begin || m_it > m_end || *(m_it - 1) != '"')
556 return false;
557
558 if (escaped) {
559 if (fromScratch != nullptr)
560 *fromScratch = true;
561 out = m_parseScratch.view();
562 } else {
563 if (fromScratch != nullptr)
564 *fromScratch = false;
565 out = json_str_view(begin, (uint32_t)((m_it - 1) - begin));
566 }
567 return true;
568 }

597 bool
598 parse_integer_token(bool& negative, uint64_t& magnitude, bool& overflow, bool& adjusted, bool& integerToken) {
599 ws();
600 negative = false;
601 magnitude = 0;
602 overflow = false;
603 adjusted = false;
604 integerToken = false;
605 if (m_it == nullptr || m_end == nullptr || m_it >= m_end)
606 return false;
607
608 const char* p = m_it;
609 if (*p == '-') {
610 negative = true;
611 ++p;
612 }
613 if (p >= m_end || *p < '0' || *p > '9')
614 return false;
615 const char* digitsBegin = p;
616 if (*p == '0' && p + 1 < m_end && p[1] >= '0' && p[1] <= '9')
617 return false;
618
619 constexpr uint64_t MaxValue = (std::numeric_limits<uint64_t>::max)();
620 while (p < m_end && *p >= '0' && *p <= '9') {
621 const auto digit = (uint64_t)(*p - '0');
622 if (magnitude > (MaxValue - digit) / 10)
623 overflow = true;
624 else if (!overflow)
625 magnitude = magnitude * 10 + digit;
626 ++p;
627 }
628 const char* digitsEnd = p;
629
630 if (p < m_end && *p == '.') {
631 const char* decimalIt = p + 1;
632 if (decimalIt >= m_end || *decimalIt < '0' || *decimalIt > '9')
633 return false;
634 while (decimalIt < m_end && *decimalIt >= '0' && *decimalIt <= '9')
635 ++decimalIt;
636 if (decimalIt < m_end && (*decimalIt == 'e' || *decimalIt == 'E')) {
637 ++decimalIt;
638 if (decimalIt < m_end && (*decimalIt == '+' || *decimalIt == '-'))
639 ++decimalIt;
640 if (decimalIt >= m_end || *decimalIt < '0' || *decimalIt > '9')
641 return false;
642 }
643 return true;
644 }
645 if (p < m_end && (*p == 'e' || *p == 'E')) {
646 const char* exponentIt = p + 1;
647 bool negativeExponent = false;
648 if (exponentIt < m_end && (*exponentIt == '+' || *exponentIt == '-')) {
649 negativeExponent = *exponentIt == '-';
650 ++exponentIt;
651 }
652 if (exponentIt >= m_end || *exponentIt < '0' || *exponentIt > '9')
653 return false;
654
655 uint32_t exponent = 0;
656 bool exponentOverflow = false;
657 while (exponentIt < m_end && *exponentIt >= '0' && *exponentIt <= '9') {
658 const auto digit = (uint32_t)(*exponentIt - '0');
659 if (exponent > (UINT32_MAX - digit) / 10)
660 exponentOverflow = true;
661 else if (!exponentOverflow)
662 exponent = exponent * 10 + digit;
663 ++exponentIt;
664 }
665
666 if (negativeExponent && (exponentOverflow || exponent != 0)) {
667 const auto digitCount = (uint32_t)(digitsEnd - digitsBegin);
668 const auto keptCount = exponentOverflow || exponent >= digitCount ? 0U : digitCount - exponent;
669 magnitude = 0;
670 overflow = false;
671 for (uint32_t i = 0; i < keptCount; ++i) {
672 const auto digit = (uint64_t)(digitsBegin[i] - '0');
673 if (magnitude > (MaxValue - digit) / 10)
674 overflow = true;
675 else if (!overflow)
676 magnitude = magnitude * 10 + digit;
677 }
678 for (uint32_t i = keptCount; i < digitCount; ++i)
679 adjusted = adjusted || digitsBegin[i] != '0';
680 } else if (exponentOverflow || exponent > 19) {
681 overflow = magnitude != 0;
682 } else {
683 while (exponent-- > 0) {
684 if (magnitude > MaxValue / 10) {
685 overflow = true;
686 break;
687 }
688 magnitude *= 10;
689 }
690 }
691 p = exponentIt;
692 }
693
694 if (overflow)
695 magnitude = MaxValue;
696
697 integerToken = true;
698 m_it = p;
699 return true;
700 }

702 bool parse_number(double& value) {
703 ws();

705 return false;

709 if (pEnd == m_it)
710 return false;
711 m_it = pEnd;
712 return true;

730
731 bool skip_value() {
732 ws();
733 if (m_it == nullptr || m_end == nullptr || m_it >= m_end)
734 return false;
735
736 if (*m_it == '{') {
737 ++m_it;
738 ws();
739 if (consume('}'))
740 return true;
741
742 while (true) {
743 json_str_view key;
744 if (!parse_string_view(key))
745 return false;
746 if (!expect(':'))
747 return false;
748 if (!skip_value())
749 return false;
750
751 ws();
752 if (consume(','))
753 continue;
754 if (consume('}'))
755 return true;
756 return false;
757 }
758 }
759
760 if (*m_it == '[') {
761 ++m_it;
762 ws();
763 if (consume(']'))
764 return true;
765
766 while (true) {
767 if (!skip_value())
768 return false;
769 ws();
770 if (consume(','))
771 continue;
772 if (consume(']'))
773 return true;
774 return false;
775 }
776 }
777
778 if (*m_it == '"') {
779 json_str_view tmp;
780 return parse_string_view(tmp);
781 }
782
783 if (*m_it == 't' || *m_it == 'f') {
784 bool v = false;
785 return parse_bool(v);
786 }
787
788 if (*m_it == 'n')
789 return parse_null();
790
791 double v = 0.0;
792 return parse_number(v);
793 }

803 template <typename TInt>
804 inline bool read_runtime_field_json_int(ser_json& reader, uint8_t* pFieldData, uint32_t size, bool& ok) {
805 bool negative = false;
806 uint64_t magnitude = 0;
807 bool overflow = false;
808 bool adjusted = false;
809 bool integerToken = false;
810 if (!reader.parse_integer_token(negative, magnitude, overflow, adjusted, integerToken))
811 return false;
812
813 if (integerToken) {
814 if (adjusted)
815 ok = false;
816 TInt value = 0;
817 if constexpr (std::is_signed_v<TInt>) {
818 constexpr auto MaxPositive = (uint64_t)(std::numeric_limits<TInt>::max)();
819 constexpr auto MaxNegative = MaxPositive + 1;
820 if (negative) {
821 if (overflow || magnitude > MaxNegative) {
822 value = (std::numeric_limits<TInt>::lowest)();
823 ok = false;
824 } else if (magnitude == MaxNegative) {
825 value = (std::numeric_limits<TInt>::lowest)();
826 } else {
827 value = (TInt) - (int64_t)magnitude;
828 }
829 } else if (overflow || magnitude > MaxPositive) {
830 value = (std::numeric_limits<TInt>::max)();
831 ok = false;
832 } else {
833 value = (TInt)magnitude;
834 }
835 } else {
836 constexpr auto MaxValue = (uint64_t)(std::numeric_limits<TInt>::max)();
837 if (negative) {
838 if (overflow || magnitude != 0)
839 ok = false;
840 } else if (overflow || magnitude > MaxValue) {
841 value = (std::numeric_limits<TInt>::max)();
842 ok = false;
843 } else {
844 value = (TInt)magnitude;
845 }
846 }
847
848 copy_field_bytes(pFieldData, size, value);
849 return true;
850 }
851
852 double d = 0.0;
853 if (!reader.parse_number(d))
854 return false;
855
856 if (!std::isfinite(d)) {
857 ok = false;
858 const TInt v = 0;
859 copy_field_bytes(pFieldData, size, v);
860 return true;
861 }
862
863 double clamped = std::trunc(d);
864 if (clamped != d)
865 ok = false;
866
867 constexpr auto minVal = (double)(std::numeric_limits<TInt>::lowest)();
868 constexpr auto maxVal = (double)(std::numeric_limits<TInt>::max)();
869 if (clamped < minVal) {
870 clamped = minVal;
871 ok = false;
872 } else if constexpr (sizeof(TInt) == sizeof(uint64_t)) {
873 if (clamped >= maxVal) {

1018 }

1020 return read_runtime_field_json_float<float>(reader, pFieldData, size, ok);

1025 case serialization_type_id::b: {

1029 copy_field_bytes(pFieldData, size, v);

1038 return true;

1042 const auto maxLen = size > 0 ? size - 1 : 0;

1057 template <typename TByteSink>
1058 inline bool parse_json_byte_array(ser_json& reader, TByteSink& out) {
1059 if (!reader.expect('['))
1060 return false;
1061
1062 reader.ws();
1063 if (reader.consume(']'))
1064 return true;
1065
1066 while (true) {
1067 double d = 0.0;
1068 if (!reader.parse_number(d))
1069 return false;
1070 if (d < 0.0 || d > 255.0)
1071 return false;
1072
1073 const auto v = (uint32_t)d;
1074 if ((double)v != d)
1075 return false;
1076
1077 const uint8_t byte = (uint8_t)v;
1078 out.save_raw(&byte, 1, serialization_type_id::u8);
1079
1080 reader.ws();
1081 if (reader.consume(','))
1082 continue;
1083 if (reader.consume(']'))
1084 return true;
1085 return false;
1086 }
