303 template <typename T>
304 constexpr T closest_pow2(T number) {
305 static_assert(std::is_integral<T>::value, "closest_pow2 must be used with integer types");
306
307 if (is_pow2(number))
308 return number;
309
310 // Collapse all bits below the highest set bit
311 number |= number >> 1;
312 number |= number >> 2;
313 number |= number >> 4;
314 if constexpr (sizeof(T) > 1)
315 number |= number >> 8;
316 if constexpr (sizeof(T) > 2)
317 number |= number >> 16;
318 if constexpr (sizeof(T) > 4)
319 number |= number >> 32;
320
321 // The result is now one less than the next power of two, so shift back
322 return number - (number >> 1);
323 }

1702 child = right;

1704 if (!cmpFunc(arr[(uint32_t)root], arr[(uint32_t)child]))
1705 return;
