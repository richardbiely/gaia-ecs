86 static void op_storage(Op op, JobCallable* dst, JobCallable* src, [[maybe_unused]] const void* pArg) {
87 auto*& pStorage = *reinterpret_cast<void**>(dst->m_storage);
88 switch (op) {
89 case Op::Invoke:
90 GAIA_ASSERT(false && "The callable was already destroyed by finish()");
91 break;
92 case Op::Finish:
93 break;
94 case Op::Destroy:
95 mem::AllocHelper::free(pStorage);
96 break;
97 case Op::Move:
98 GAIA_ASSERT(src != nullptr);
99 pStorage = *reinterpret_cast<void**>(src->m_storage);
100 dst->m_func = src->m_func;
101 src->m_func = nullptr;
102 break;
103 }
104 }

136 template <typename F>
137 void init(F&& f) {
138 using Fn = std::decay_t<F>;
139 static_assert(
140 !std::is_same_v<Fn, util::SmallFunc> && !detail::is_move_func<Fn>::value,
141 "Pass the callable itself. SmallFunc and MoveFunc may keep it in SmallBlockAllocator, which job "
142 "threads must not free from");
143 static_assert(invocable<Fn>(), "JobCallable requires a callable compatible with its signature");
144 static_assert(std::is_move_constructible_v<Fn>, "Callable must be move-constructible");
145 static_assert(
146 alignof(Fn) <= alignof(std::max_align_t), "Over-aligned callables are not supported for JobCallable");
147
148 if constexpr (stores_inline<Fn>()) {
149 new (m_storage) Fn(GAIA_FWD(f));
150
151 m_func = [](Op op, JobCallable* dst, JobCallable* src, const void* pArg) {
152 auto* pFn = reinterpret_cast<Fn*>(dst->m_storage);
153 switch (op) {
154 case Op::Invoke:
155 invoke(*pFn, pArg);
156 break;
157 case Op::Finish:
158 // A trivially destructible callable has nothing to end and is left untouched.
159 if constexpr (!std::is_trivially_destructible_v<Fn>) {
160 pFn->~Fn();
161 dst->m_func = nullptr;
162 }
163 break;
164 case Op::Destroy:
165 if constexpr (!std::is_trivially_destructible_v<Fn>)
166 pFn->~Fn();
167 break;
168 case Op::Move: {
169 GAIA_ASSERT(src != nullptr);
170 auto* pSrcFn = reinterpret_cast<Fn*>(src->m_storage);
171 new (dst->m_storage) Fn(GAIA_MOV(*pSrcFn));
172 if constexpr (!std::is_trivially_destructible_v<Fn>)
173 pSrcFn->~Fn();
174 dst->m_func = src->m_func;
175 src->m_func = nullptr;
176 break;
177 }
178 }
179 };
180 } else {
181 auto* pStorage = mem::AllocHelper::alloc<Fn>();
182 GAIA_ASSERT((uintptr_t)pStorage % alignof(Fn) == 0);
183 auto* pFunc = new (pStorage) Fn(GAIA_FWD(f));
184 *reinterpret_cast<Fn**>(m_storage) = pFunc;
185
186 m_func = [](Op op, JobCallable* dst, JobCallable* src, const void* pArg) {
187 auto*& pFn = *reinterpret_cast<Fn**>(dst->m_storage);
188 switch (op) {
189 case Op::Invoke:
190 GAIA_ASSERT(pFn != nullptr);
191 invoke(*pFn, pArg);
192 break;
193 case Op::Finish:
194 // A trivially destructible callable has nothing to end and is left untouched.
195 if constexpr (!std::is_trivially_destructible_v<Fn>) {
196 GAIA_ASSERT(pFn != nullptr);
197 pFn->~Fn();
198 dst->m_func = &op_storage;
199 }
200 break;
201 case Op::Destroy:
202 GAIA_ASSERT(pFn != nullptr);
203 if constexpr (!std::is_trivially_destructible_v<Fn>)
204 pFn->~Fn();
205 mem::AllocHelper::free(pFn);
206 break;
207 case Op::Move:
208 GAIA_ASSERT(src != nullptr);
209 *reinterpret_cast<Fn**>(dst->m_storage) = *reinterpret_cast<Fn**>(src->m_storage);
210 dst->m_func = src->m_func;
211 *reinterpret_cast<Fn**>(src->m_storage) = nullptr;
212 src->m_func = nullptr;
213 break;
214 }
215 };
216 }
217 }
