268 sarr_soa(core::zero_t) noexcept {
269 // explicit zeroing
270 for (auto i = (size_type)0; i < extent; ++i)
271 operator[](i) = {};

280 template <typename InputIt>
281 sarr_soa(InputIt first, InputIt last) noexcept {
282 const auto count = (size_type)core::distance(first, last);
283
284 if constexpr (std::is_pointer_v<InputIt>) {
285 for (size_type i = 0; i < count; ++i)
286 operator[](i) = first[i];
287 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
288 for (size_type i = 0; i < count; ++i)
289 operator[](i) = *(first[i]);
290 } else {
291 size_type i = 0;
292 for (auto it = first; it != last; ++it)
293 operator[](i++) = *it;
294 }
295 }
