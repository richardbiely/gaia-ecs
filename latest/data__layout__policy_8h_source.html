31 template <typename T>
32 constexpr uint32_t get_alignment() {
33 if constexpr (std::is_empty_v<T>)
34 // Always consider 0 for empty types
35 return 0U;
36 else {
37 // Use at least 4 (32-bit systems) or 8 (64-bit systems) bytes for alignment
38 return (uint32_t)core::get_min(sizeof(uintptr_t), alignof(T));
39 }

48 constexpr size_t get_aligned_byte_offset(uintptr_t address, size_t alig, size_t itemSize, size_t cnt) {
49 const auto padding = mem::padding(address, alig);
50 address += padding + itemSize * cnt;
51 return address;

62 const auto padding = mem::padding<Alignment>(address);

618 template <typename Tup, size_t... Ids>
619 GAIA_NODISCARD constexpr static ValueType
620 get_inter(Tup&& t, std::span<const uint8_t> s, size_t idx, std::index_sequence<Ids...> /*no_name*/) noexcept {
621 auto address = mem::align<Alignment>((uintptr_t)s.data());
622 ((
623 // Put the value at the address into our tuple. Data is aligned so we can read directly.
624 std::get<Ids>(t) = get_ref<value_type<Ids>>((const uint8_t*)address, idx),
625 // Skip towards the next element and make sure the address is aligned properly
626 address = mem::align<Alignment>(address + (sizeof(value_type<Ids>) * s.size()))),
627 ...);
628 return meta::tuple_to_struct<ValueType, TTuple>(GAIA_FWD(t));
629 }

635 ((
636 // Set the tuple value. Data is aligned so we can write directly.
637 get_ref<value_type<Ids>>((uint8_t*)address, idx) = std::get<Ids>(t),
