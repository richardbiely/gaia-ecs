616 template <typename Tup, size_t... Ids>
617 GAIA_NODISCARD constexpr static ValueType
618 get_inter(Tup&& t, std::span<const uint8_t> s, size_t idx, std::index_sequence<Ids...> /*no_name*/) noexcept {
619 auto address = mem::align<Alignment>((uintptr_t)s.data());
620 ((
621 // Put the value at the address into our tuple. Data is aligned so we can read directly.
622 std::get<Ids>(t) = get_ref<value_type<Ids>>((const uint8_t*)address, idx),
623 // Skip towards the next element and make sure the address is aligned properly
624 address = mem::align<Alignment>(address + (sizeof(value_type<Ids>) * s.size()))),
625 ...);
626 return meta::tuple_to_struct<ValueType, TTuple>(GAIA_FWD(t));
627 }

633 ((
634 // Set the tuple value. Data is aligned so we can write directly.
