84 template <typename InputIt>
85 constexpr sarr(InputIt first, InputIt last) noexcept: sarr() {
86 const auto count = (size_type)core::distance(first, last);
87
88 if constexpr (std::is_pointer_v<InputIt>) {
89 for (size_type i = 0; i < count; ++i)
90 operator[](i) = first[i];
91 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
92 for (size_type i = 0; i < count; ++i)
93 operator[](i) = *(first[i]);
94 } else {
95 size_type i = 0;
96 for (auto it = first; it != last; ++it)
97 operator[](i++) = *it;
98 }
99 }
