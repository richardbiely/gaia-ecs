98 template <typename InputIt>
99 constexpr sarr_ext(InputIt first, InputIt last) noexcept {
100 const auto count = (size_type)core::distance(first, last);
101 resize(count);
102
103 if constexpr (std::is_pointer_v<InputIt>) {
104 for (size_type i = 0; i < count; ++i)
105 operator[](i) = first[i];
106 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
107 for (size_type i = 0; i < count; ++i)
108 operator[](i) = *(first[i]);
109 } else {
110 size_type i = 0;
111 for (auto it = first; it != last; ++it)
112 operator[](i++) = *it;
113 }
114 }

295 constexpr iterator erase(iterator pos) noexcept {
296 GAIA_ASSERT(pos >= data());
297 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
298
299 if (empty())
300 return end();
301
302 const auto idxSrc = (size_type)core::distance(begin(), pos);
303 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
304
305 mem::shift_elements_left<T, false>(m_data, idxDst, idxSrc, extent);
306 // Destroy if it's the last element
307 auto* ptr = &data()[m_cnt - 1];
308 core::call_dtor(ptr);
309
310 --m_cnt;
311
312 return iterator(&data()[idxSrc]);
313 }

319 iterator erase(iterator first, iterator last) noexcept {
320 GAIA_ASSERT(first >= data())
321 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
322 GAIA_ASSERT(last > first);
323 GAIA_ASSERT(last <= (data() + size()));
324
325 if (empty())
326 return end();
327
328 const auto idxSrc = (size_type)core::distance(begin(), first);
329 const auto idxDst = size();
330 const auto cnt = (size_type)(last - first);
331
332 mem::shift_elements_left_fast<T, false>(m_data, idxDst, idxSrc, cnt, extent);
333 // Destroy if it's the last element
334 core::call_dtor_n(&data()[m_cnt - cnt], cnt);
335
336 m_cnt -= cnt;
337
338 return iterator(&data()[idxSrc]);
339 }

366 constexpr void resize(size_type count) noexcept {
367 GAIA_ASSERT(count <= max_size());
368
369 // Resizing to a smaller size
370 if (count <= m_cnt) {
371 // Destroy elements at the end
372 core::call_dtor_n(&data()[count], size() - count);
373 } else
374 // Resizing to a bigger size but still within allocated capacity
375 {
376 // Construct new elements
377 core::call_ctor_n(&data()[size()], count - size());
378 }
379
380 m_cnt = count;
381 }

386 constexpr void resize(size_type count, const_reference value) noexcept {
387 const auto oldCount = m_cnt;
388 resize(count);
389
390 if constexpr (std::is_copy_constructible_v<value_type>) {
391 const value_type valueCopy = value;
392 for (size_type i = oldCount; i < m_cnt; ++i)
393 operator[](i) = valueCopy;
394 } else {
395 for (size_type i = oldCount; i < m_cnt; ++i)
396 operator[](i) = value;
397 }
398 }

404 template <typename Func>
405 auto retain(Func&& func) noexcept {
406 size_type erased = 0;
407 size_type idxDst = 0;
408 size_type idxSrc = 0;
409
410 while (idxSrc < m_cnt) {
411 if (func(operator[](idxSrc))) {
412 if (idxDst < idxSrc) {
413 auto* ptr = (uint8_t*)data();
414 mem::move_element<T, false>(ptr, ptr, idxDst, idxSrc, max_size(), max_size());
415 auto* ptr2 = &data()[idxSrc];
416 core::call_dtor(ptr2);
417 }
418 ++idxDst;
419 } else {
420 auto* ptr = &data()[idxSrc];
421 core::call_dtor(ptr);
422 ++erased;
423 }
424
425 ++idxSrc;
426 }
427
428 m_cnt -= erased;
429 return idxDst;
430 }
