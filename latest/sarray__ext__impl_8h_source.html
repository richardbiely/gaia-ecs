99 template <typename InputIt>
100 constexpr sarr_ext(InputIt first, InputIt last) noexcept {
101 const auto count = (size_type)core::distance(first, last);
102 resize(count);
103
104 if constexpr (std::is_pointer_v<InputIt>) {
105 for (size_type i = 0; i < count; ++i)
106 operator[](i) = first[i];
107 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
108 for (size_type i = 0; i < count; ++i)
109 operator[](i) = *(first[i]);
110 } else {
111 size_type i = 0;
112 for (auto it = first; it != last; ++it)
113 operator[](i++) = *it;
114 }
115 }

253 iterator insert(iterator pos, const T& arg) noexcept {
254 GAIA_ASSERT(size() < N);
255 GAIA_ASSERT(pos >= data());
256 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
257
258 const auto idxSrc = (size_type)core::distance(begin(), pos);
259 const auto idxDst = (size_type)core::distance(begin(), end());
260
261 T value(arg);
262 auto* ptr = &data()[idxSrc];
263 if (idxSrc == idxDst) {
264 core::call_ctor(ptr, std::move_if_noexcept(value));
265 } else {
266 mem::move_ctor_elements<T>((uint8_t*)&data()[idxDst], (uint8_t*)&data()[idxDst - 1], 1);
267 mem::shift_elements_right<T, false>(m_data, idxDst - 1, idxSrc, extent);
268 mem::move_element<T, false>((uint8_t*)ptr, (uint8_t*)&value, 0, 0, 1, 1);
269 }
270
271 ++m_cnt;
272
273 return iterator(&data()[idxSrc]);
274 }

306 constexpr iterator erase(iterator pos) noexcept {
307 GAIA_ASSERT(pos >= data());
308 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
309
310 if (empty())
311 return end();
312
313 const auto idxSrc = (size_type)core::distance(begin(), pos);
314 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
315
316 mem::shift_elements_left<T, false>(m_data, idxDst, idxSrc, extent);
317 // Destroy if it's the last element
318 auto* ptr = &data()[m_cnt - 1];
319 core::call_dtor(ptr);
320
321 --m_cnt;
322
323 return iterator(&data()[idxSrc]);
324 }
325
330 iterator erase(iterator first, iterator last) noexcept {
331 GAIA_ASSERT(first >= data())
332 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
333 GAIA_ASSERT(last > first);
334 GAIA_ASSERT(last <= (data() + size()));
335
336 if (empty())
337 return end();
338
339 const auto idxSrc = (size_type)core::distance(begin(), first);
340 const auto idxDst = size();
341 const auto cnt = (size_type)(last - first);
342
343 mem::shift_elements_left_fast<T, false>(m_data, idxDst, idxSrc, cnt, extent);
344 // Destroy if it's the last element
345 core::call_dtor_n(&data()[m_cnt - cnt], cnt);
346
347 m_cnt -= cnt;
348
349 return iterator(&data()[idxSrc]);
350 }

377 constexpr void resize(size_type count) noexcept {
378 GAIA_ASSERT(count <= max_size());
379
380 // Resizing to a smaller size
381 if (count <= m_cnt) {
382 // Destroy elements at the end
383 core::call_dtor_n(&data()[count], size() - count);
384 } else
385 // Resizing to a bigger size but still within allocated capacity
386 {
387 // Construct new elements
388 core::call_ctor_n(&data()[size()], count - size());
389 }
390
391 m_cnt = count;
392 }

397 constexpr void resize(size_type count, const_reference value) noexcept {
398 const auto oldCount = m_cnt;
399 resize(count);
400
401 if constexpr (std::is_copy_constructible_v<value_type>) {
402 const value_type valueCopy = value;
403 for (size_type i = oldCount; i < m_cnt; ++i)
404 operator[](i) = valueCopy;
405 } else {
406 for (size_type i = oldCount; i < m_cnt; ++i)
407 operator[](i) = value;
408 }
