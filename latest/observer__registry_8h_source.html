55 GAIA_NODISCARD size_t hash() const {
56 size_t seed = EntityLookupKey(bindingRelation).hash();
57 seed ^= EntityLookupKey(traversalRelation).hash() + 0x9e3779b9u + (seed << 6u) + (seed >> 2u);
58 seed ^= EntityLookupKey(rootTarget).hash() + 0x9e3779b9u + (seed << 6u) + (seed >> 2u);

61 return seed;
62 }

452 explicit CandidateScope(ObserverRegistry& registry): m_registry(registry) {
453 const auto depth = m_registry.m_candidate_depth++;
454 if (depth == 0) {
455 m_pCandidates = &m_registry.m_candidate_observers;

459 m_registry.m_nested_candidate_observers.push_back(new cnt::darray<ObserverRuntimeData*>());
460 m_pCandidates = m_registry.m_nested_candidate_observers[nestedIdx];
461 }
462 m_pCandidates->clear();
463 }
