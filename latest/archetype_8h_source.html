362 void update_data_offsets(uintptr_t memoryAddress) {
363 uintptr_t offset = 0;
364
365 // Versions
366 // We expect versions to fit in the first 256 bytes.
367 // With 32 components per archetype this gives us some headroom.
368 {
369 offset += mem::padding<alignof(ComponentVersion)>(memoryAddress);
370
371 const auto cnt = (uint32_t)m_shape.properties.cntEntities + 1U; // + 1 for entities
372 GAIA_ASSERT(offset < 256);
373 m_shape.dataOffsets.firstByte_Versions = (ChunkDataVersionOffset)offset;
374 offset += sizeof(ComponentVersion) * cnt;
375 }
376
377 // Entity ids
378 {
379 offset += mem::padding<alignof(Entity)>(offset);
380
381 const auto cnt = m_shape.properties.cntEntities;
382 if (cnt != 0) {
383 m_shape.dataOffsets.firstByte_CompEntities = (ChunkDataOffset)offset;
384
385 // Storage-wise, treat the component array as it it were MAX_COMPONENTS long.
386 offset += sizeof(Entity) * ChunkHeader::MAX_COMPONENTS;
387 }
388 }
389
390 // Component records
391 {
392 offset += mem::padding<alignof(ComponentRecord)>(offset);
393
394 const auto cnt = m_shape.properties.cntEntities;
395 if (cnt != 0) {
396
397 m_shape.dataOffsets.firstByte_Records = (ChunkDataOffset)offset;
398
399 // Storage-wise, treat the component array as it it were MAX_COMPONENTS long.
400 offset += sizeof(ComponentRecord) * cnt;
401 }
402 }
403
404 // First entity offset
405 {
406 offset += mem::padding<alignof(Entity)>(offset);
407 m_shape.dataOffsets.firstByte_EntityData = (ChunkDataOffset)offset;
408 }
409 }

419 static bool est_max_entities_per_chunk(
420 uint32_t offs, const Entity* ids, const ComponentCacheItem* const* pItems, uint32_t cnt, uint32_t cap,
421 uint32_t maxDataOffset) {
422 GAIA_FOR(cnt) {
423 const auto comp = comp_from_item(ids[i], pItems[i]);
424 if (!component_uses_table_storage(comp))
425 continue;
426
427 const auto* pItem = pItems[i];
428 GAIA_ASSERT(pItem != nullptr);
429
430 // If we're beyond what the chunk could take, subtract one entity
431 offs = pItem->calc_new_mem_offset(offs, cap);
432 if (offs >= maxDataOffset)
433 return false;
434 }
435
436 return true;
437 }

439 static void reg_components(
440 Archetype& arch, EntitySpan ids, const ComponentCacheItem* const* pItems, uint8_t from, uint8_t to,
441 uint32_t& currOff, uint32_t count) {
442 auto& ofs = arch.m_shape.compOffs;
443
444 // Set component ids
445 GAIA_FOR2(from, to) arch.m_shape.ids[i] = ids[i];
446
447 // Calculate offsets and assign them indices according to our mappings
448 GAIA_FOR2(from, to) {
449 const auto comp = comp_from_item(ids[i], pItems[i]);
450 const auto compIdx = i;
451
452 if (!component_uses_table_storage(comp)) {
453 ofs[compIdx] = {};
454 } else {
455 const auto alig = comp.alig();
456 currOff = mem::align(currOff, alig);
457 ofs[compIdx] = (ChunkDataOffset)currOff;
458
459 currOff = pItems[i]->calc_new_mem_offset(currOff, count);
460 }
461 }

538 GAIA_NODISCARD static Archetype*
539 create(const World& world, ArchetypeId archetypeId, uint32_t& worldVersion, EntitySpan ids) {
540 const auto& cc = comp_cache(world);
541
542 auto* newArch = mem::AllocHelper::alloc<Archetype>("Archetype");
543 (void)new (newArch) Archetype(world, cc, worldVersion);
544
545 newArch->m_archetypeId = archetypeId;
546 newArch->m_shape.archetypeIdHash = ArchetypeIdLookupKey::calc(archetypeId);
547
548 // Calculate component mask. This will be used to early exit matching archetypes in simple queries.
549 // TODO: Performance could be improved if we're an archetype comming from one already known.
550 // We could simply take the predecessor's mask and update it with the new ids.
551 newArch->m_shape.queryMask = build_entity_mask({ids.data(), ids.size()});
552
553 const auto cnt = (uint32_t)ids.size();
554 newArch->m_shape.properties.cntEntities = (uint8_t)ids.size();
555
556 auto compItems = std::span(&newArch->m_shape.compItems[0], cnt);
557 GAIA_FOR(cnt) compItems[i] = ids[i].pair() ? cc.find_pair_payload(ids[i]) : cc.find(ids[i]);
558
559 // Calculate offsets
560 static auto ChunkDataAreaOffset = Chunk::chunk_data_area_offset();
561 newArch->update_data_offsets(
562 // This is not a real memory address.
563 // Chunk memory is organized as header+data. The offsets we calculate here belong to
564 // the data area.
565 // Every allocated chunk is going to have the same relative offset from the header part
566 // which is why providing a fictional relative offset is enough.
567 ChunkDataAreaOffset);
568 const auto& offs = newArch->m_shape.dataOffsets;
569 newArch->m_pPairIndex = PairIndexData::create(ids);
570
571 uint32_t compsSize = 0;
572 GAIA_FOR(cnt)
573 compsSize += comp_from_item(ids[i], newArch->m_shape.compItems[i]).size();
574
575 auto compute_max_entities_for_chunk = [&](uint32_t maxEntities, uint32_t dataLimit) -> uint32_t {
576 uint32_t low = 1;
577 uint32_t high = maxEntities;
578 uint32_t best = 1;
579
580 // Helper to test if a given entity count fits in the chunk
581 auto try_fit = [&](uint32_t count) -> bool {
582 const uint32_t currOff = offs.firstByte_EntityData + (count * sizeof(Entity));
583 return est_max_entities_per_chunk(currOff, ids.data(), newArch->m_shape.compItems, cnt, count, dataLimit);
584 };
585
586 // Binary search for the lookup
587 while (low <= high) {
588 uint32_t mid = (low + high) / 2;
589 if (try_fit(mid)) {
590 best = mid;
591 low = mid + 1;
592 } else {
593 high = mid - 1;
594 }
595 }
596
597 return best;
598 };
599
600 // Calculate the number of entities per chunk precisely so we can fit as many of them into a
601 // chunk as possible. Start at the smallest size class and only upsize when the archetype is
602 // wide enough that a smaller chunk can't hold the target row count.
603 constexpr uint32_t MinEntitiesPerChunk = 1536;
604 uint32_t maxItemsInArchetype = 0;
605
606 auto compute_max_entities_for_size_type = [&](uint32_t sizeType) -> uint32_t {
607 const uint32_t dataLimit = Chunk::chunk_data_bytes(mem_block_size(sizeType));
608 const uint32_t fixedSize = offs.firstByte_EntityData + 1;
609 GAIA_ASSERT(dataLimit > fixedSize);
610
611 // Theoretical maximum number of component rows we can fit into one chunk.
612 // This can be further reduced due to alignment and padding.
613 const uint32_t itemSize = compsSize + (uint32_t)sizeof(Entity);
614 const uint32_t maxEntities = (dataLimit - fixedSize) / itemSize;
615 return compute_max_entities_for_chunk(maxEntities > 0 ? maxEntities : 1, dataLimit);
616 };
617
618 if (archetypeId == 0) {
619 // Keep the root archetype compact enough to avoid growing the maximum row snapshot used by iterators.
620 maxItemsInArchetype = compute_max_entities_for_size_type(2);
621 } else {
622 for (uint32_t sizeType = 0; sizeType < MemoryBlockSizeClasses; ++sizeType) {
623 maxItemsInArchetype = compute_max_entities_for_size_type(sizeType);
624 if (maxItemsInArchetype >= MinEntitiesPerChunk)
625 break;
626 }
627 }
628
629 // MAX_CHUNK_ENTITIES is intentionally based on the 32 KiB class to keep iterator snapshots bounded.
630 // Wide archetypes can still use the 64 KiB-class blocks because their row count stays below this cap.
631 if (maxItemsInArchetype > ChunkHeader::MAX_CHUNK_ENTITIES)
632 maxItemsInArchetype = ChunkHeader::MAX_CHUNK_ENTITIES;
633
634 // Update the offsets according to the recalculated maxItemsInArchetype
635 auto currOff = offs.firstByte_EntityData + ((uint32_t)sizeof(Entity) * maxItemsInArchetype);
636 reg_components(
637 *newArch, ids, newArch->m_shape.compItems, (uint8_t)0, (uint8_t)ids.size(), currOff, maxItemsInArchetype);
638
639 newArch->m_shape.properties.capacity = (uint16_t)maxItemsInArchetype;
640 newArch->m_shape.properties.chunkDataBytes = (ChunkDataOffset)currOff;
641
642 return newArch;
643 }

683 void del(Chunk* pChunk) {
684 // Make sure there are any chunks to delete
685 GAIA_ASSERT(!m_storage.chunks.empty());
686
687 const auto chunkIndex = pChunk->idx();
688
689 // Make sure the chunk is a part of the chunk array
690 GAIA_ASSERT(chunkIndex == core::get_index(m_storage.chunks, pChunk));
691
692 // Remove the chunk from the chunk array. We are swapping this chunk's entry
693 // with the last one in the array. Therefore, we first update the last item's
694 // index with the current chunk's index and then do the swapping.
695 m_storage.chunks.back()->set_idx(chunkIndex);
696 core::swap_erase(m_storage.chunks, chunkIndex);
697
698 // Delete the chunk now. Otherwise, if the chunk happened to be the last
699 // one we would end up overriding released memory.
700 Chunk::free(pChunk);
701 }

707 GAIA_NODISCARD Chunk* foc_free_chunk() {
708 const auto chunkCnt = m_storage.chunks.size();
709
710 if (chunkCnt > 0) {
711 for (uint32_t i = m_storage.firstFreeChunkIdx; i < m_storage.chunks.size(); ++i) {
712 auto* pChunk = m_storage.chunks[i];
713 GAIA_ASSERT(pChunk != nullptr);
714 const auto entityCnt = pChunk->size();
715 if (entityCnt < pChunk->capacity()) {
716 m_storage.firstFreeChunkIdx = i;
717 return pChunk;
718 }
719 }
720 }
721
722 // Make sure not too many chunks are allocated
723 GAIA_ASSERT(chunkCnt < UINT32_MAX);
724
725 // No free space found anywhere. Let's create a new chunk.
726 auto* pChunk = Chunk::create(
727 m_world, m_cc, chunkCnt, //
728 m_shape.properties.capacity, m_shape.properties.cntEntities, //
729 m_shape.properties.chunkDataBytes, //
730 m_worldVersion, m_shape.dataOffsets, m_shape.ids, m_shape.compItems, m_shape.compOffs);
731
732 m_storage.firstFreeChunkIdx = m_storage.chunks.size();
733 m_storage.chunks.push_back(pChunk);
734 return pChunk;
735 }

989 void sort_entities_inter(size_t low, size_t high, TSortByFunc func) {
990 while (low < high) {
991 const auto pivotIdx = low + (high - low) / 2;
992 Entity pivotEntity = get_flat_entity<Enabled>(pivotIdx);
993 if (pivotIdx != high) {
994 const Entity highEntity = get_flat_entity<Enabled>(high);
995 Chunk::swap_chunk_entities(const_cast<World&>(m_world), pivotEntity, highEntity);
996 }
997
998 size_t i = low;
999 for (size_t j = low; j < high; ++j) {
1000 Entity jEntity = get_flat_entity<Enabled>(j);
1001 if (func(m_world, &jEntity, &pivotEntity) < 0) {
1002 if (i != j) {
1003 Entity iEntity = get_flat_entity<Enabled>(i);
1004 Chunk::swap_chunk_entities(const_cast<World&>(m_world), iEntity, jEntity);
1005 }
1006 ++i;
1007 }

1010 {
1011 Entity iEntity = get_flat_entity<Enabled>(i);
1012 Chunk::swap_chunk_entities(const_cast<World&>(m_world), iEntity, pivotEntity);
1013 }
1014
1017 if (i - low < high - i) {
1018 if (low < i)
1019 sort_entities_inter<Enabled>(low, i - 1, func);
1020 low = i + 1;
1021 } else {
