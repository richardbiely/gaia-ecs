109 template <typename InputIt>
110 darr_ext(InputIt first, InputIt last) {
111 const auto count = (size_type)core::distance(first, last);
112 resize(count);
113
114 if constexpr (std::is_pointer_v<InputIt>) {
115 for (size_type i = 0; i < count; ++i)
116 operator[](i) = first[i];
117 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
118 for (size_type i = 0; i < count; ++i)
119 operator[](i) = *(first[i]);
120 } else {
121 size_type i = 0;
122 for (auto it = first; it != last; ++it)
123 operator[](i++) = *it;
124 }
125 }

269

278 core::call_dtor_n(data(), m_cnt);
279 GAIA_MEM_SANI_DEL_BLOCK(value_size, m_data, m_cap, m_cnt);

288 void resize(size_type count) {
289 if (count == m_cnt)
290 return;
291
292 // Resizing to a smaller size
293 if (count < m_cnt) {
294 // Destroy elements at the end
295 core::call_dtor_n(&data()[count], m_cnt - count);
296 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, m_cnt - count);
297
298 m_cnt = count;
299 return;
300 }
301
302 // Resizing to a bigger size but still within allocated capacity
303 if (count <= m_cap) {
304 // Construct new elements
305 GAIA_MEM_SANI_PUSH_N(value_size, data(), m_cap, m_cnt, count - m_cnt);
306 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
307
308 m_cnt = count;
309 return;
310 }
311
312 auto* pDataOld = m_pDataHeap;
313 m_pDataHeap = view_policy::template alloc<Allocator>(count);
314 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pDataHeap, count, count);
315 auto* pDataNew = reinterpret_cast<pointer>(m_pDataHeap);
316 if (pDataOld != nullptr) {
317 mem::move_ctor_elements<T>(m_pDataHeap, pDataOld, m_cnt);
318 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
319 } else {
320 mem::move_ctor_elements<T>(m_pDataHeap, m_data, m_cnt);
321 core::call_dtor_n(data(), m_cnt);
322 GAIA_MEM_SANI_DEL_BLOCK(value_size, m_data, m_cap, m_cnt);
323 }
324
325 core::call_ctor_n(&pDataNew[m_cnt], count - m_cnt);
326
327 m_cap = count;
328 m_cnt = count;
329 m_pData = m_pDataHeap;
330 }

335 void resize(size_type count, const_reference value) {

371 template <typename... Args>
372 decltype(auto) emplace_back(Args&&... args) {
373 if GAIA_UNLIKELY (m_cnt == m_cap) {
374 const auto cap = m_cap != 0 ? (m_cap * 3 + 1) / 2 : 4;
375 auto* pDataNew = view_policy::template alloc<Allocator>(cap);
376 GAIA_MEM_SANI_ADD_BLOCK(value_size, pDataNew, cap, m_cnt + 1);
378 core::call_ctor(&static_cast<T*>(static_cast<void*>(pDataNew))[m_cnt], GAIA_FWD(args)...);
379 mem::move_ctor_elements<T>(pDataNew, m_pData, m_cnt);
380 if (m_pDataHeap != nullptr)
381 view_policy::template free<Allocator>(m_pDataHeap, m_cap, m_cnt);
382 else {
383 core::call_dtor_n(data(), m_cnt);
384 GAIA_MEM_SANI_DEL_BLOCK(value_size, m_data, m_cap, m_cnt);
385 }
386 m_pDataHeap = pDataNew;
387 m_pData = pDataNew;
388 m_cap = cap;
389 } else {
390 GAIA_MEM_SANI_PUSH(value_size, m_pData, m_cap, m_cnt);
391 core::call_ctor(&data()[m_cnt], GAIA_FWD(args)...);
392 }
393 return (reference)data()[m_cnt++];
394 }

411 iterator insert(iterator pos, const T& arg) {
412 GAIA_ASSERT(pos >= data());
413 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
414
415 T value(arg);
416 const auto idxSrc = (size_type)core::distance(begin(), pos);
417 try_grow();
418 const auto idxDst = (size_type)core::distance(begin(), end());
419
420 GAIA_MEM_SANI_PUSH(value_size, data(), m_cap, m_cnt);
421 auto* ptr = &data()[idxSrc];
422 if (idxSrc == idxDst) {
423 core::call_ctor(ptr, std::move_if_noexcept(value));
424 } else {
425 mem::move_ctor_elements<T>((uint8_t*)&data()[idxDst], (uint8_t*)&data()[idxDst - 1], 1);
426 mem::shift_elements_right<T, false>(m_pData, idxDst - 1, idxSrc, m_cap);
427 mem::move_element<T, false>((uint8_t*)ptr, (uint8_t*)&value, 0, 0, 1, 1);
428 }
429
430 ++m_cnt;
431
432 return iterator(ptr);
433 }

466 iterator erase(iterator pos) noexcept {
467 GAIA_ASSERT(pos >= data());
468 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
469
470 if (empty())
471 return end();
472
473 const auto idxSrc = (size_type)core::distance(begin(), pos);
474 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
475
476 mem::shift_elements_left<T, false>(m_pData, idxDst, idxSrc, m_cap);
477 // Destroy if it's the last element
478 auto* ptr = &data()[m_cnt - 1];
479 core::call_dtor(ptr);
480 GAIA_MEM_SANI_POP(value_size, data(), m_cap, m_cnt);
481
482 --m_cnt;
483
484 return iterator(&data()[idxSrc]);
485 }

491 iterator erase(iterator first, iterator last) noexcept {
492 GAIA_ASSERT(first >= data())
493 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
494 GAIA_ASSERT(last > first);
495 GAIA_ASSERT(last <= iterator(data() + size()));
496
497 if (empty())
498 return end();
499
500 const auto idxSrc = (size_type)core::distance(begin(), first);
501 const auto idxDst = size();
502 const auto cnt = (size_type)(last - first);
503
504 mem::shift_elements_left_fast<T, false>(m_pData, idxDst, idxSrc, cnt, m_cap);
505 // Destroy if it's the last element
506 core::call_dtor_n(&data()[m_cnt - cnt], cnt);
507 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, cnt);
508
509 m_cnt -= cnt;
510
511 return iterator(&data()[idxSrc]);
512 }

520 void shrink_to_fit() {
521 const auto cap = capacity();
522 const auto cnt = size();
523 if (m_pDataHeap == nullptr || (cnt > extent && cap == cnt))
524 return;
525
526 auto* pDataOld = m_pDataHeap;
527 if (cnt <= extent) {
528 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_data, extent, cnt);
529 mem::move_ctor_elements<T>(m_data, pDataOld, cnt);
530 m_pDataHeap = nullptr;
531 m_pData = m_data;
532 m_cap = extent;
533 } else {
534 m_pDataHeap = view_policy::template alloc<Allocator>(cnt);
535 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pDataHeap, cnt, cnt);
536 mem::move_ctor_elements<T>(m_pDataHeap, pDataOld, cnt);
537 m_pData = m_pDataHeap;
538 m_cap = cnt;
539 }
540 view_policy::template free<Allocator>(pDataOld, cap, cnt);
541 }
