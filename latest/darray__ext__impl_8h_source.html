81 void try_grow() {
82 const auto cnt = size();
83 const auto cap = capacity();
84
85 // Unless we reached the capacity don't do anything
86 if GAIA_LIKELY (cnt < cap)
87 return;
88
89 // We increase the capacity in multiples of 1.5 which is about the golden ratio (1.618).
90 // This means we prefer more frequent allocations over memory fragmentation.
91 m_cap = (cap * 3 + 1) / 2;
92
93 if GAIA_UNLIKELY (m_pDataHeap == nullptr) {
94 // If no heap memory is allocated yet we need to allocate it and move the old stack elements to it
95 m_pDataHeap = view_policy::template alloc<Allocator>(m_cap);
96 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pDataHeap, m_cap, cnt);
97 mem::move_elements<T, false>(m_pDataHeap, m_data, cnt, 0, m_cap, cap);
98 } else {
99 // Move items from the old heap array to the new one. Delete the old
100 auto* pDataOld = m_pDataHeap;
101 m_pDataHeap = view_policy::template alloc<Allocator>(m_cap);
102 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pDataHeap, m_cap, cnt);
103 mem::move_elements<T, false>(m_pDataHeap, pDataOld, cnt, 0, m_cap, cap);
104 view_policy::template free<Allocator>(pDataOld, cap, cnt);
105 }
106
107 m_pData = m_pDataHeap;
108 }

133 darr_ext(InputIt first, InputIt last) {
134 const auto count = (size_type)core::distance(first, last);
135 resize(count);
136
137 if constexpr (std::is_pointer_v<InputIt>) {
138 for (size_type i = 0; i < count; ++i)

140 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
141 for (size_type i = 0; i < count; ++i)
142 operator[](i) = *(first[i]);
143 } else {
144 size_type i = 0;
145 for (auto it = first; it != last; ++it)
146 operator[](i++) = *it;
147 }

284 if (cap <= m_cap)
285 return;
286

288 m_pDataHeap = view_policy::template alloc<Allocator>(cap);

295 GAIA_MEM_SANI_DEL_BLOCK(value_size, m_data, m_cap, m_cnt);
296 }

304 void resize(size_type count) {
305 if (count == m_cnt)
306 return;
307
308 // Resizing to a smaller size
309 if (count < m_cnt) {
310 // Destroy elements at the end
311 core::call_dtor_n(&data()[count], m_cnt - count);
312 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, m_cnt - count);
313
314 m_cnt = count;
315 return;
316 }
317
318 // Resizing to a bigger size but still within allocated capacity
319 if (count <= m_cap) {
320 // Construct new elements
321 GAIA_MEM_SANI_PUSH_N(value_size, data(), m_cap, m_cnt, count - m_cnt);
322 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
323
324 m_cnt = count;
325 return;
326 }
327
328 auto* pDataOld = m_pDataHeap;
329 m_pDataHeap = view_policy::template alloc<Allocator>(count);
330 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pDataHeap, count, count);
331 auto* pDataNew = reinterpret_cast<pointer>(m_pDataHeap);
332 if (pDataOld != nullptr) {
333 mem::move_elements<T, false>(m_pDataHeap, pDataOld, m_cnt, 0, count, m_cap);
334 core::call_ctor_n(&pDataNew[m_cnt], count - m_cnt);
335 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
336 } else {
337 mem::move_elements<T, false>(m_pDataHeap, m_data, m_cnt, 0, count, m_cap);
338 GAIA_MEM_SANI_DEL_BLOCK(value_size, m_data, m_cap, m_cnt);
339 }
340
341 m_cap = count;
342 m_cnt = count;
343 m_pData = m_pDataHeap;
344 }

455 iterator erase(iterator pos) noexcept {
456 GAIA_ASSERT(pos >= data());
457 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
458
459 if (empty())
460 return end();
461
462 const auto idxSrc = (size_type)core::distance(begin(), pos);
463 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
464

480 iterator erase(iterator first, iterator last) noexcept {
481 GAIA_ASSERT(first >= data())
482 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
483 GAIA_ASSERT(last > first);
484 GAIA_ASSERT(last <= iterator(data() + size()));
485
486 if (empty())
487 return end();
488

490 const auto idxDst = size();
491 const auto cnt = (size_type)(last - first);
492
493 mem::shift_elements_left_fast<T, false>(m_pData, idxDst, idxSrc, cnt, m_cap);

496 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, cnt);
497
498 m_cnt -= cnt;
499

509 void shrink_to_fit() {
510 const auto cap = capacity();
511 const auto cnt = size();
512
513 if (cap == cnt)
514 return;
515
516 if (m_pDataHeap != nullptr) {
517 auto* pDataOld = m_pDataHeap;
518
519 if (cnt < extent) {
520 mem::move_elements<T, false>(m_data, pDataOld, cnt, 0);
521 m_pData = m_data;
522 m_cap = extent;
523 } else {
524 m_pDataHeap = view_policy::template alloc<Allocator>(m_cap = cnt);
525 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pDataHeap, m_cap, m_cnt);
526 mem::move_elements<T, false>(m_pDataHeap, pDataOld, cnt, 0);
527 m_pData = m_pDataHeap;
528 }
529
530 GAIA_MEM_SANI_DEL_BLOCK(value_size, pDataOld, cap, cnt);
531 view_policy::template free<Allocator>(pDataOld);
532 } else
533 resize(cnt);
534 }

540 template <typename Func>
541 auto retain(Func&& func) noexcept {
542 size_type erased = 0;
543 size_type idxDst = 0;
544 size_type idxSrc = 0;
545
546 while (idxSrc < m_cnt) {
547 if (func(operator[](idxSrc))) {
548 if (idxDst < idxSrc) {
549 auto* ptr = (uint8_t*)data();
550 mem::move_element<T, false>(ptr, ptr, idxDst, idxSrc, m_cap, m_cap);
551 auto* ptr2 = &data()[idxSrc];
552 core::call_dtor(ptr2);
553 }
554 ++idxDst;
555 } else {
556 auto* ptr = &data()[idxSrc];
557 core::call_dtor(ptr);
558 ++erased;
559 }
560
561 ++idxSrc;
562 }
563
564 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, erased);
565
566 m_cnt -= erased;
567 return idxDst;
568 }
