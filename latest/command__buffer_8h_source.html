365 GAIA_NODISCARD Entity resolve(Entity e) const {
366 if (!is_tmp(e))
367 return e;
368
369 const auto ti = e.id();
370 if (ti < m_temp2real.size())
371 return m_temp2real[ti];
372
373 return EntityBad;
374 }

379 GAIA_NODISCARD Entity resolve_object(const Op& op) const {
380 if (op.pairTarget == EntityBad)
381 return resolve(op.other);
382
383 const auto relation = resolve(op.other);
384 const auto target = resolve(op.pairTarget);
385 if (relation == EntityBad || target == EntityBad)
386 return EntityBad;
387
388 return (Entity)Pair(relation, target);
389 }

395 void allocate_instantiate_group(uint32_t firstIdx) {
396 const Op& first = m_ops[firstIdx];
397 const Entity prefab = resolve(first.other);
398 if (prefab == EntityBad)
399 return;
400
401 Entity parent = EntityBad;
402 if (first.pairTarget != EntityBad) {
403 parent = resolve(first.pairTarget);
404 if (parent == EntityBad)
405 return;
406 }
407
408 cnt::darray_ext<uint32_t, 16> tempIds;
409 for (uint32_t j = firstIdx; j < m_ops.size(); ++j) {
410 const Op& op = m_ops[j];
411 if (op.type != OpType::INSTANTIATE_ENTITY || op.other != first.other || op.pairTarget != first.pairTarget)
412 continue;
413 if (!is_tmp(op.target))
414 continue;
415
416 const uint32_t ti = op.target.id();
417 if (ti >= m_temp2real.size() || m_temp2real[ti] != EntityBad || is_canceled_temp(ti))
418 continue;
419
420 tempIds.push_back(ti);
421 }
422
423 if (tempIds.empty())
424 return;
425
426 if (tempIds.size() == 1) {
427 m_temp2real[tempIds[0]] =
428 parent == EntityBad ? m_world.instantiate(prefab) : m_world.instantiate(prefab, parent);
429 return;
430 }
431
432 uint32_t mapped = 0;
433 auto mapRoot = [&](Entity instance) {
434 GAIA_ASSERT(mapped < tempIds.size());
435 m_temp2real[tempIds[mapped++]] = instance;
436 };
437 if (parent == EntityBad)
438 m_world.instantiate_n(prefab, tempIds.size(), mapRoot);
439 else
440 m_world.instantiate_n(prefab, parent, tempIds.size(), mapRoot);
441 GAIA_ASSERT(mapped == tempIds.size());
442 }

447 void replay_add(Entity target, Entity object) {
448 if (object.pair()) {
449 World::EntityBuilder(m_world, target).add(Pair(m_world.get(object.id()), m_world.get(object.gen())));
450 return;
451 }
452
453 const auto* pItem = m_world.comp_cache().find(object);
454 if (pItem != nullptr && pItem->comp.storage_type() == DataStorageType::Sparse) {
455 const auto mode = m_world.sparse_storage_mode(object);
456 GAIA_ASSERT(mode != World::SparseStorageMode::None);
457 auto& store = m_world.sparse_component_store_erased_mut(object, *pItem);
458 (void)store.func_add(store.pStore, target);
459 m_world.finish_sparse_component_add_inter(target, object, mode);
460 return;
461 }
462
463 World::EntityBuilder(m_world, target).add(object);
464 }

481 void replay_data(Entity target, Entity object, uint32_t dataPos, bool finishWrite = true) {
482 auto serializer = ser::make_serializer(m_data);
483 serializer.seek(dataPos);
484 const auto* pItem =
485 object.pair() ? m_world.comp_cache().find_pair_payload(object) : m_world.comp_cache().find(object);
486 GAIA_ASSERT(pItem != nullptr);
487 if (pItem == nullptr)
488 return;
489 const auto& item = *pItem;
490
491 if (m_world.component_uses_sparse_storage(object)) {
492 const auto payload = m_world.mut_raw(target, object);
493 GAIA_ASSERT(payload.valid());
494 if (payload.valid())
495 item.load(serializer, payload.data, 0, 1, 1);
496 if (finishWrite)
497 m_world.finish_write(target, object);
498 return;
499 }
500
501 const auto& ec = target.pair() ? m_world.fetch(target) : m_world.m_recs.entities[target.id()];
502 const auto row = ec.row;
503 const auto compIdx = ec.pChunk->comp_idx(object);
504 auto* pComponentData = (void*)ec.pChunk->comp_ptr_mut(compIdx, 0);
505 item.load(serializer, pComponentData, row, row + 1, ec.pChunk->capacity());
506 if (finishWrite)
507 m_world.finish_write(target, object);
508 }

514 void replay_add_data(Entity target, Entity object, uint32_t dataPos) {
515 const auto* pItem =
516 object.pair() ? m_world.comp_cache().find_pair_payload(object) : m_world.comp_cache().find(object);
517 GAIA_ASSERT(pItem != nullptr);
518 if (pItem == nullptr)
519 return;
520 const auto& item = *pItem;
521 if (m_world.component_uses_sparse_storage(object)) {
522 const auto mode = m_world.sparse_storage_mode(object);
523 GAIA_ASSERT(mode != World::SparseStorageMode::None);
524 auto& store = m_world.sparse_component_store_erased_mut(object, item);
525 auto* pPayload = store.func_add(store.pStore, target);
526
527 auto serializer = ser::make_serializer(m_data);
528 serializer.seek(dataPos);
529 item.load(serializer, pPayload, 0, 1, 1);
530 m_world.finish_sparse_component_add_inter(target, object, mode);
531 return;
532 }
533
534 World::EntityBuilder builder(m_world, target);
535#if GAIA_OBSERVERS_ENABLED
536 auto addDiffCtx = m_world.m_observers.prepare_diff(
537 m_world, ObserverEvent::OnAdd, EntitySpan{&object, 1}, EntitySpan{&target, 1});
538#endif
539 builder.add_inter_init(object);
540 builder.commit();
541 replay_data(target, object, dataPos, false);
542 m_world.notify_add_single(target, object);
543#if GAIA_OBSERVERS_ENABLED
544 m_world.m_observers.finish_diff(m_world, GAIA_MOV(addDiffCtx));
545#endif
546 }

552 GAIA_NODISCARD bool is_canceled_temp(uint32_t idx) const {
553 const uint32_t base = idx * 3;
554 if (base + 2 >= m_tmpFlags.size())
555 return false;
556
557 const bool destroy = m_tmpFlags.test(base);
558 const bool usedOther = m_tmpFlags.test(base + 2);
559
560 // If deleted in this batch and not referenced by others, cancel entirely
561 return destroy && !usedOther;
562 }

566 GAIA_NODISCARD Entity add_temp() {
567 m_temp2real.push_back(EntityBad);
568 Entity tmp(m_nextTemp++, 0, true, false);
569 // Use the unused flag to mark temporary entities
570 tmp.data.tmp = 1;
571 return tmp;
572 }

578 GAIA_NODISCARD static bool less_op_key(const Op& a, const Op& b) {
579 if (a.target != b.target)
580 return a.target < b.target;
581
582 const bool aIsPair = a.pairTarget != EntityBad;
583 const bool bIsPair = b.pairTarget != EntityBad;
584 if (aIsPair != bIsPair)
585 return !aIsPair;
586 if (aIsPair)
587 return false;
588
589 if (a.other != b.other)
590 return a.other < b.other;
591 return false;
592 }

596 void check_sort(const Op& op) {
597 if (!m_ops.empty() && less_op_key(op, m_ops.back()))
598 m_needsSort = true;
599 }

603 void push_op(Op&& op) {
604 op.order = m_ops.size();
605 check_sort(op);
606 m_ops.push_back(GAIA_MOV(op));
607 }

621 public:
625 void commit() {
626 core::lock_scope lock(m_acc);
627
628 if (m_ops.empty())
629 return;
630
631 GAIA_PROF_SCOPE(cmdbuf::commit);
632
633 // Build flags + allocate entities
634 if (m_nextTemp > 0) {
635 GAIA_PROF_SCOPE(cmdbuf::alloc);
636
637 // Bit layout for each temporary entity:
638 // bit 0 -> marked for destruction (DEL_ENTITY recorded)
639 // bit 1 -> used as a relation target (appeared as target in component ops)
640 // bit 2 -> used as a relation source / dependency (appeared as other in component ops)
641 m_tmpFlags.resize(m_nextTemp * 3);
642
643 // Pre-map surviving temps for ADD/CPY/INSTANTIATE (avoid allocating canceled temps)
644 if (m_temp2real.size() < m_nextTemp) {
645 const auto from = m_temp2real.size();
646 m_temp2real.resize(m_nextTemp);
647 GAIA_FOR2(from, m_nextTemp) m_temp2real[i] = EntityBad;
648 }
649
650 // Build all flags
651 for (const Op& o: m_ops) {
652 // Set flag bits
653 if (is_tmp(o.target)) {
654 const uint32_t ti = o.target.id();
655
656 if (o.type == OpType::DEL_ENTITY)
657 m_tmpFlags.set((ti * 3) + 0, true);
658 else if (is_rel(o.type))
659 m_tmpFlags.set((ti * 3) + 1, true);
660 }
661
662 if (is_tmp(o.other) && o.other.id() < m_tmpFlags.size())
663 m_tmpFlags.set((o.other.id() * 3) + 2, true);
664 if (is_tmp(o.pairTarget) && o.pairTarget.id() < m_tmpFlags.size())
665 m_tmpFlags.set((o.pairTarget.id() * 3) + 2, true);
666 }
667
668 // Allocate real entities for the surviving temporaries
669 for (uint32_t i = 0; i < m_ops.size(); ++i) {
670 const Op& o = m_ops[i];
671 if (!is_tmp(o.target))
672 continue;
673
674 const uint32_t ti = o.target.id();
675 if (is_canceled_temp(ti) || m_temp2real[ti] != EntityBad)
676 continue;
677
678 if (o.type == OpType::ADD_ENTITY) {
679 m_temp2real[ti] = m_world.add();
680 } else if (o.type == OpType::CPY_ENTITY) {
681 const Entity src = resolve(o.other);
682 if (src != EntityBad)
683 m_temp2real[ti] = m_world.copy(src);
684 } else if (o.type == OpType::INSTANTIATE_ENTITY) {
685 allocate_instantiate_group(i);
686 }
687 }
688 }
689
690 // Sort by (target, other), reduce last-wins, apply relations.
691 // DEL_COMPONENT last per target.
692 if (m_needsSort) {
693 GAIA_PROF_SCOPE(cmdbuf::sort);
694
695 m_needsSort = false;
696 core::sort(m_ops.begin(), m_ops.end(), [](const Op& a, const Op& b) {
697 if (less_op_key(a, b))
698 return true;
699 if (less_op_key(b, a))
700 return false;
701 return a.order < b.order;
702 });
703 }
704
705 // Replay batched operations
706 Entity lastKey = EntityBad;
707 Entity lastResolved = EntityBad;
708 auto resolve_cached = [&](Entity e) {
709 if (e == lastKey)
710 return lastResolved;
711 lastKey = e;
712 return lastResolved = resolve(e);
713 };
714
715 cnt::darray_ext<Entity, 16> deleteTargets;
716 {
717 GAIA_PROF_SCOPE(cmdbuf::merges);
718 for (uint32_t p = 0; p < m_ops.size();) {
719 GAIA_PROF_SCOPE(cmdbuf::merge);
720
721 const Entity tgtKey = m_ops[p].target;
722
723 const bool tgtIsTemp = is_tmp(tgtKey);
724 const uint32_t ti = tgtIsTemp ? tgtKey.id() : 0u;
725 const Entity tgtReal =
726 tgtIsTemp ? (ti < m_temp2real.size() ? m_temp2real[ti] : EntityBad) : resolve_cached(tgtKey);
727
728 // Range for this target
729 uint32_t q = p;
730 bool hasDelEntity = false;
731 while (q < m_ops.size() && m_ops[q].target == tgtKey) {
732 if (m_ops[q].type == OpType::DEL_ENTITY)
733 hasDelEntity = true;
734 ++q;
735 }
736
737 // Skip canceled or non-existent temporary entities
738 if (tgtReal == EntityBad) {
739 p = q;
740 continue;
741 }
742 if (tgtIsTemp && is_canceled_temp(ti)) {
743 p = q;
744 continue;
745 }
746
747 enum : uint8_t { F_ADD = 1 << 0, F_ADD_DATA = 1 << 1, F_SET = 1 << 2, F_DEL = 1 << 3 };
748
749 // Emit relation groups.
750 // Inside [p..q) range (same target), process groups by 'other'.
751 // We perform per-component reduction.
752 for (uint32_t i = p; i < q;) {
753 const Entity othKey = m_ops[i].other;
754 const Entity pairTargetKey = m_ops[i].pairTarget;
755 const Entity othReal = resolve_object(m_ops[i]);
756
757 // Group ops with same (target, other)
758 uint32_t j = i + 1;
759 while (j < q && m_ops[j].other == othKey && m_ops[j].pairTarget == pairTargetKey)
760 ++j;
761
762 if (tgtReal != EntityBad) {
763 const uint32_t groupSize = j - i;
764 // Fast path - single op
765 if (groupSize == 1) {
766 const Op& op = m_ops[i];
767 switch (op.type) {
768 case OpType::DEL_COMPONENT:
769 replay_del(tgtReal, othReal);
770 break;
771 case OpType::ADD_COMPONENT:
772 replay_add(tgtReal, othReal);
773 break;
774 case OpType::ADD_COMPONENT_DATA:
775 replay_add_data(tgtReal, othReal, op.off);
776 break;
777 case OpType::SET_COMPONENT:
778 replay_data(tgtReal, othReal, op.off);
779 break;
780 default:
781 break;
782 }
783 }
784 // Slow path: merge multiple ops
785 else {
786 uint8_t mask = 0;
787 uint32_t dataPos = 0;
788
789 for (uint32_t k = i; k < j; ++k) {
790 const Op& op = m_ops[k];
791 switch (op.type) {
792 case OpType::ADD_COMPONENT:
793 mask |= F_ADD;
794 break;
795 case OpType::ADD_COMPONENT_DATA:
796 mask |= F_ADD_DATA;
797 dataPos = op.off;
798 break;
799 case OpType::SET_COMPONENT:
800 mask |= F_SET;
801 dataPos = op.off;
802 break;
803 case OpType::DEL_COMPONENT:
804 mask |= F_DEL;
805 break;
806 default:
807 break;
808 }
809 }
810
811 const bool hasAdd = mask & F_ADD;
812 const bool hasAddData = mask & F_ADD_DATA;
813 const bool hasSet = mask & F_SET;
814 const bool hasDel = mask & F_DEL;
815
816 // 1) ADD(+DATA) + DEL = no-op
817 if (hasDel && (hasAdd || hasAddData)) {
818 }
819 // 2) DEL only
820 else if (hasDel) {
821 replay_del(tgtReal, othReal);
822 }
823 // 3) ADD_WITH_DATA or ADD+SET = ADD_WITH_DATA
824 else if (hasAddData || (hasAdd && hasSet)) {
825 replay_add_data(tgtReal, othReal, dataPos);
826 }
827 // 4) ADD only
828 else if (hasAdd) {
829 replay_add(tgtReal, othReal);
830 }
831 // 5) SET only
832 else if (hasSet) {
833 replay_data(tgtReal, othReal, dataPos);
834 }
835 }
836 }
837
838 // Advance to next component group
839 i = j;
840 }
841
842 // Apply deletions after every relationship operation so sorting target groups cannot
843 // invalidate an endpoint that a later group still needs to resolve.
844 if (hasDelEntity)
845 deleteTargets.push_back(tgtReal);
846
847 // Advance to next target group
848 p = q;
849 }
850 }
851 for (auto entity: deleteTargets) {
852 if (m_world.valid(entity))
853 m_world.del(entity);
854 }
855
856 clear();
857 }
858 };
