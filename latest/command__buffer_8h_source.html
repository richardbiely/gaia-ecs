502 GAIA_NODISCARD Entity resolve_object(const Op& op) const {
503 if (op.pairTarget == EntityBad)
504 return resolve(op.other);
505
506 const auto relation = resolve(op.other);
507 const auto target = resolve(op.pairTarget);
508 if (relation == EntityBad || target == EntityBad)
509 return EntityBad;
510
511 return (Entity)Pair(relation, target);

518 void allocate_instantiate_group(uint32_t firstIdx) {
519 const Op& first = m_ops[firstIdx];
520 const Entity prefab = resolve(first.other);
521 if (prefab == EntityBad)
522 return;
523
524 Entity parent = EntityBad;
525 if (first.pairTarget != EntityBad) {
526 parent = resolve(first.pairTarget);
527 if (parent == EntityBad)
528 return;
529 }
530
531 cnt::darray_ext<uint32_t, 16> tempIds;
532 for (uint32_t j = firstIdx; j < m_ops.size(); ++j) {
533 const Op& op = m_ops[j];
534 if (op.type != OpType::INSTANTIATE_ENTITY || op.other != first.other || op.pairTarget != first.pairTarget)
535 continue;
536 if (!is_tmp(op.target))
537 continue;
538
539 const uint32_t ti = op.target.id();
540 if (ti >= m_temp2real.size() || m_temp2real[ti] != EntityBad || is_canceled_temp(ti))
541 continue;
542
543 tempIds.push_back(ti);
544 }
545
546 if (tempIds.empty())
547 return;
548
549 if (tempIds.size() == 1) {
550 m_temp2real[tempIds[0]] =
551 parent == EntityBad ? m_world.instantiate(prefab) : m_world.instantiate(prefab, parent);
552 return;
553 }
554
555 uint32_t mapped = 0;
556 auto mapRoot = [&](Entity instance) {
557 GAIA_ASSERT(mapped < tempIds.size());
558 m_temp2real[tempIds[mapped++]] = instance;
559 };
560 if (parent == EntityBad)
561 m_world.instantiate_n(prefab, tempIds.size(), mapRoot);
562 else
563 m_world.instantiate_n(prefab, parent, tempIds.size(), mapRoot);
564 GAIA_ASSERT(mapped == tempIds.size());
565 }

570 void replay_add(Entity target, Entity object) {
571 if (object.pair()) {
572 World::EntityBuilder(m_world, target).add(Pair(m_world.get(object.id()), m_world.get(object.gen())));
573 return;
574 }
575
576 const auto* pItem = m_world.comp_cache().find(object);
577 if (pItem != nullptr && pItem->comp.storage_type() == DataStorageType::Sparse) {
578 const auto mode = m_world.sparse_storage_mode(object);
579 GAIA_ASSERT(mode != World::SparseStorageMode::None);
580 auto& store = m_world.sparse_component_store_erased_mut(object, *pItem);
581 (void)store.func_add(store.pStore, target);
582 m_world.finish_sparse_component_add_inter(target, object, mode);
583 return;
584 }
585
586 World::EntityBuilder(m_world, target).add(object);
587 }

605 template <bool Reset = false>
606 void replay_data(Entity target, Entity object, uint32_t dataPos, bool finishWrite = true) {
607 auto serializer = ser::make_serializer(m_data);
608 serializer.seek(dataPos);
609 const auto* pItem =
610 object.pair() ? m_world.comp_cache().find_pair_payload(object) : m_world.comp_cache().find(object);
611 GAIA_ASSERT(pItem != nullptr);
612 if (pItem == nullptr)
613 return;
614 const auto& item = *pItem;
615
616 if (m_world.component_uses_sparse_storage(object)) {
617 const auto payload = m_world.mut_raw(target, object);
618 GAIA_ASSERT(payload.valid());
619 if (payload.valid()) {
620 if constexpr (Reset) {
621 if (item.func_ctor != nullptr)
622 item.func_ctor(payload.data, 1);
623 }
624 item.load(serializer, payload.data, 0, 1, 1);
625 }
626 if (finishWrite)
627 m_world.finish_write(target, object);
628 return;
629 }
630
631 const auto& ec = target.pair() ? m_world.fetch(target) : m_world.m_recs.entities[target.id()];
632 const auto row = ec.row;
633 const auto compIdx = ec.pChunk->comp_idx(object);
634 if constexpr (Reset)
635 ec.pChunk->call_ctor(row, compIdx, item);
636 auto* pComponentData = (void*)ec.pChunk->comp_ptr_mut(compIdx, 0);
637 item.load(serializer, pComponentData, row, row + 1, ec.pChunk->capacity());
638 if (finishWrite)
639 m_world.finish_write(target, object);
640 }

646 void replay_add_data(Entity target, Entity object, uint32_t dataPos) {
647 const auto* pItem =
648 object.pair() ? m_world.comp_cache().find_pair_payload(object) : m_world.comp_cache().find(object);
649 GAIA_ASSERT(pItem != nullptr);
650 if (pItem == nullptr)
651 return;
652 const auto& item = *pItem;
653 if (gaia::ecs::component_uses_sparse_storage(item.comp)) {
654 const auto mode = m_world.sparse_storage_mode(object);
655 GAIA_ASSERT(mode != World::SparseStorageMode::None);
656 auto& store = m_world.sparse_component_store_erased_mut(object, item);
657 auto* pPayload = store.func_add(store.pStore, target);
658
659 auto serializer = ser::make_serializer(m_data);
660 serializer.seek(dataPos);
661 item.load(serializer, pPayload, 0, 1, 1);
662 m_world.finish_sparse_component_add_inter(target, object, mode);
663 return;
664 }
665
666 World::EntityBuilder builder(m_world, target);
667#if GAIA_OBSERVERS_ENABLED
668 auto addDiffCtx = m_world.m_observers.prepare_diff(
669 m_world, ObserverEvent::OnAdd, EntitySpan{&object, 1}, EntitySpan{&target, 1});
670#endif
671 builder.add_inter_init(object);
672 builder.commit();
673 replay_data(target, object, dataPos, false);
674 m_world.notify_add_single(target, object);
675#if GAIA_OBSERVERS_ENABLED
676 m_world.m_observers.finish_diff(m_world, GAIA_MOV(addDiffCtx));
677#endif
678 }

684 void replay_op(const Op& op, Entity target, Entity object) {
685 switch (op.type) {
686 case OpType::DEL_COMPONENT:
687 replay_del(target, object);
688 break;
689 case OpType::ADD_COMPONENT:
690 replay_add(target, object);
691 break;
692 case OpType::ADD_COMPONENT_DATA:
693 replay_add_data(target, object, op.off);
694 break;
695 case OpType::SET_COMPONENT:
696 replay_data(target, object, op.off);
697 break;
698 default:
699 break;
700 }
701 }

711 GAIA_NOINLINE void
712 replay_merged(uint32_t i, uint32_t j, Entity target, Entity object, uint8_t mask, uint32_t dataPos) {
713 // Reduce ordinary component lifetimes when intermediate mutations have no ECS side effects.
714 bool canReduce = !object.pair() && object.id() > GAIA_ID(LastCoreComponent).id() &&
715 !m_world.m_hasRequiresPolicy && !m_world.m_hasCantCombinePolicy &&
716 !m_world.has_direct(object, Requires);
717#if GAIA_OBSERVERS_ENABLED
718 canReduce = canReduce && !m_world.m_observers.has_on_add_observers() &&
719 !m_world.m_observers.has_on_del_observers() && !m_world.m_observers.has_on_set_observers(object);
720#endif
721 const bool initial = canReduce && m_world.has_direct(target, object);
722 const bool hasAdd = (mask & (F_ADD | F_ADD_DATA)) != 0;
723 const auto* pItem = canReduce ? m_world.comp_cache().find(object) : nullptr;
724 if (pItem != nullptr) {
725 // Tags with constructors need World's chunk-backed construction path.
726 canReduce = pItem->func_dtor == nullptr && (pItem->comp.size() != 0 || pItem->func_ctor == nullptr);
727#if GAIA_ENABLE_ADD_DEL_HOOKS
728 canReduce = canReduce && pItem->hooks().func_add == nullptr && pItem->hooks().func_del == nullptr;
729#endif
730#if GAIA_ENABLE_SET_HOOKS
731 canReduce = canReduce && pItem->hooks().func_set == nullptr;
732#endif
733 }
734 if (canReduce) {
735 const bool hasData = (mask & (F_ADD_DATA | F_SET)) != 0;
736 GAIA_ASSERT(!hasData || hasAdd);
737 if (!hasAdd) {
738 if (initial)
739 replay_del(target, object);
740 } else if (hasData) {
741 if (initial)
742 replay_data<true>(target, object, dataPos);
743 else
744 replay_add_data(target, object, dataPos);
745 } else if (initial) {
746 // Bare re-adds run their default constructor without moving the entity.
747 if (pItem != nullptr && pItem->func_ctor != nullptr) {
748 const auto payload = m_world.mut_raw(target, object);
749 GAIA_ASSERT(payload.valid());
750 pItem->func_ctor(payload.data, 1);
751 m_world.finish_write(target, object);
752 }
753 } else {
754 replay_add(target, object);
755 }
756 return;
757 }
758
759 for (uint32_t k = i; k < j; ++k) {
760 const Op& op = m_ops[k];
761 // Merged groups may contain redundant deletes of an absent component.
762 if (op.type != OpType::DEL_COMPONENT || m_world.has_direct(target, object))
763 replay_op(op, target, object);
764 }
765 }

773 GAIA_NOINLINE bool replay_batch(uint32_t& p, uint32_t q, Entity target) {
774 if (target.pair() || m_world.m_hasRequiresPolicy || m_world.m_hasCantCombinePolicy)
775 return false;
776#if GAIA_OBSERVERS_ENABLED
777 if (m_world.m_observers.has_on_add_observers() || m_world.m_observers.has_on_del_observers())
778 return false;
779#endif
780 struct Group {
781 Entity object;
782 uint32_t dataPos;
783 uint8_t mask;
784 bool initial;
785 };
786 cnt::sarray_ext<Group, ChunkHeader::MAX_COMPONENTS> groups;
787 bool hasStructural = false;
788 const Archetype* pArchetype = nullptr;
789 uint32_t end = p;
790 for (uint32_t i = p; i < q;) {
791 // Allocation records were already applied; entity deletion remains deferred.
792 if (m_ops[i].type < OpType::ADD_COMPONENT) {
793 ++i;
794 continue;
795 }
796 if (m_ops[i].pairTarget != EntityBad || groups.size() == groups.capacity())
797 break;
798 const Entity object = resolve_object(m_ops[i]);
799 if (object == EntityBad || object.pair() || object.id() <= GAIA_ID(LastCoreComponent).id())
800 break;
801 const auto* pItem = m_world.comp_cache().find(object);
802 if (pItem != nullptr) {
803 if (pItem->comp.storage_type() == DataStorageType::Sparse || pItem->func_dtor != nullptr ||
804 (pItem->comp.size() == 0 && pItem->func_ctor != nullptr))
805 break;
806#if GAIA_ENABLE_ADD_DEL_HOOKS
807 if (pItem->hooks().func_add != nullptr || pItem->hooks().func_del != nullptr)
808 break;
809#endif
810#if GAIA_ENABLE_SET_HOOKS
811 if (pItem->hooks().func_set != nullptr)
812 break;
813#endif
814 }
815#if GAIA_OBSERVERS_ENABLED
816 if (m_world.m_observers.has_on_set_observers(object))
817 break;
818#endif
819 if (m_world.component_is_non_fragmenting(object))
820 break;
821 if (pArchetype == nullptr)
822 pArchetype = m_world.fetch(target).pArchetype;
823 Group group{object, 0, 0, pArchetype->has(object)};
824 const Entity key = m_ops[i].other;
825 do {
826 const Op& op = m_ops[i];
827 switch (op.type) {
828 case OpType::ADD_COMPONENT:
829 group.mask |= F_ADD;
830 break;
831 case OpType::ADD_COMPONENT_DATA:
832 group.mask |= F_ADD_DATA;
833 group.dataPos = op.off;
834 break;
835 case OpType::SET_COMPONENT:
836 group.mask |= F_SET;
837 group.dataPos = op.off;
838 break;
839 case OpType::DEL_COMPONENT:
840 group.mask = F_DEL;
841 break;
842 default:
843 break;
844 }
845 ++i;
846 } while (i < q && m_ops[i].other == key && m_ops[i].pairTarget == EntityBad);
847 if ((group.mask & F_DEL) != 0 && m_world.has_direct(object, Requires))
848 break;
849 const bool hasAdd = (group.mask & (F_ADD | F_ADD_DATA)) != 0;
850 hasStructural |= hasAdd ? !group.initial : ((group.mask & F_DEL) != 0 && group.initial);
851 groups.push_back(group);
852 end = i;
853 }
854 if (groups.size() < 2)
855 return false;
856
857 if (hasStructural) {
858 World::EntityBuilder builder(m_world, target);
859 for (const auto& group: groups) {
860 const bool hasAdd = (group.mask & (F_ADD | F_ADD_DATA)) != 0;
861 if (hasAdd && !group.initial)
862 builder.add_inter_init(group.object);
863 else if (!hasAdd && (group.mask & F_DEL) != 0 && group.initial)
864 builder.del_inter(group.object);
865 }
866 builder.commit();
867 }
868
869 // Restore surviving payloads only after the destination storage is available.
870 for (const auto& group: groups) {
871 const bool hasAdd = (group.mask & (F_ADD | F_ADD_DATA)) != 0;
872 const bool hasDel = (group.mask & F_DEL) != 0;
873 if (hasDel && !hasAdd) {
874 GAIA_ASSERT((group.mask & F_SET) == 0);
875 continue;
876 }
877 if ((group.mask & (F_ADD_DATA | F_SET)) != 0) {
878 if (hasDel && group.initial)
879 replay_data<true>(target, group.object, group.dataPos);
880 else
881 replay_data(target, group.object, group.dataPos, !hasAdd);
882 } else if (hasDel && group.initial) {
883 const auto* pItem = m_world.comp_cache().find(group.object);
884 if (pItem != nullptr && pItem->func_ctor != nullptr) {
885 const auto& ec = m_world.fetch(target);
886 ec.pChunk->call_ctor(ec.row, ec.pChunk->comp_idx(group.object), *pItem);
887 m_world.finish_write(target, group.object);
888 }
889 }
890 }
891 p = end;
892 return true;
893 }

899 GAIA_NODISCARD bool is_canceled_temp(uint32_t idx) const {
900 const uint32_t base = idx * 3;
901 if (base + 2 >= m_tmpFlags.size())
902 return false;
903
904 const bool destroy = m_tmpFlags.test(base);
905 const bool usedOther = m_tmpFlags.test(base + 2);
906
907 // If deleted in this batch and not referenced by others, cancel entirely
908 return destroy && !usedOther;
909 }

925 GAIA_NODISCARD static bool less_op_key(const Op& a, const Op& b) {
926 if (a.target != b.target)
927 return a.target < b.target;
928
929 const bool aIsPair = a.pairTarget != EntityBad;
930 const bool bIsPair = b.pairTarget != EntityBad;
931 if (aIsPair != bIsPair)
932 return !aIsPair;
933 if (aIsPair)
934 return false;
935
936 if (a.other != b.other)
937 return a.other < b.other;
938 return false;
939 }

961 void replay_plain() {
962 for (const Op& op: m_ops) {
963 switch (op.type) {
964 case OpType::ADD_ENTITY:
965 m_temp2real[op.target.id()] = m_world.add();
966 break;
967 case OpType::CPY_ENTITY:
968 m_temp2real[op.target.id()] = m_world.copy(resolve(op.other));
969 break;
970 case OpType::INSTANTIATE_ENTITY:
971 m_temp2real[op.target.id()] = op.pairTarget == EntityBad ?
972 m_world.instantiate(resolve(op.other)) :
973 m_world.instantiate(resolve(op.other), resolve(op.pairTarget));
974 break;
975 case OpType::DEL_ENTITY:
976 m_world.del(resolve(op.target));
977 break;
978 default:
979 replay_op(op, resolve(op.target), resolve_object(op));
980 break;
981 }
982 }

997 public:
1008 void commit() {
1009 core::lock_scope lock(m_acc);
1010
1011 if (m_ops.empty())
1012 return;
1013
1014 GAIA_PROF_SCOPE(cmdbuf::commit);
1015
1016 if constexpr (!Optimize) {
1017 replay_plain();
1018 clear();
1019 return;
1020 }
1021
1022 // Build flags + allocate entities
1023 if (m_nextTemp > 0) {
1024 GAIA_PROF_SCOPE(cmdbuf::alloc);
1025
1026 // Bit layout for each temporary entity:
1027 // bit 0 -> marked for destruction (DEL_ENTITY recorded)
1028 // bit 1 -> used as a relation target (appeared as target in component ops)
1029 // bit 2 -> used as a relation source / dependency (appeared as other in component ops)
1030 m_tmpFlags.resize(m_nextTemp * 3);
1031
1032 // Pre-map surviving temps for ADD/CPY/INSTANTIATE (avoid allocating canceled temps)
1033 if (m_temp2real.size() < m_nextTemp) {
1034 const auto from = m_temp2real.size();
1035 m_temp2real.resize(m_nextTemp);
1036 GAIA_FOR2(from, m_nextTemp) m_temp2real[i] = EntityBad;
1037 }
1038
1039 // Build all flags
1040 for (const Op& o: m_ops) {
1041 // Set flag bits
1042 if (is_tmp(o.target)) {
1043 const uint32_t ti = o.target.id();
1044
1045 if (o.type == OpType::DEL_ENTITY)
1046 m_tmpFlags.set((ti * 3) + 0, true);
1047 else if (is_rel(o.type))
1048 m_tmpFlags.set((ti * 3) + 1, true);
1049 }
1050
1051 if (is_tmp(o.other) && o.other.id() < m_tmpFlags.size())
1052 m_tmpFlags.set((o.other.id() * 3) + 2, true);
1053 if (is_tmp(o.pairTarget) && o.pairTarget.id() < m_tmpFlags.size())
1054 m_tmpFlags.set((o.pairTarget.id() * 3) + 2, true);
1055 }
1056
1057 // Allocate real entities for the surviving temporaries
1058 for (uint32_t i = 0; i < m_ops.size(); ++i) {
1059 const Op& o = m_ops[i];
1060 if (!is_tmp(o.target))
1061 continue;
1062
1063 const uint32_t ti = o.target.id();
1064 if (is_canceled_temp(ti) || m_temp2real[ti] != EntityBad)
1065 continue;
1066
1067 if (o.type == OpType::ADD_ENTITY) {
1068 m_temp2real[ti] = m_world.add();
1069 } else if (o.type == OpType::CPY_ENTITY) {
1070 const Entity src = resolve(o.other);
1071 if (src != EntityBad)
1072 m_temp2real[ti] = m_world.copy(src);
1073 } else if (o.type == OpType::INSTANTIATE_ENTITY) {
1074 allocate_instantiate_group(i);
1075 }
1076 }
1077 }
1078
1079 // Sort by (target, other), reduce each component group, apply relations.
1080 // Preserve recorded order within each component group and each target's pair operations.
1081 if (m_needsSort) {
1082 GAIA_PROF_SCOPE(cmdbuf::sort);
1083
1084 m_needsSort = false;
1085 core::sort(m_ops.begin(), m_ops.end(), [](const Op& a, const Op& b) {
1086 if (less_op_key(a, b))
1087 return true;
1088 if (less_op_key(b, a))
1089 return false;
1090 return a.order < b.order;
1091 });
1092 }
1093
1094 // Replay batched operations
1095 Entity lastKey = EntityBad;
1096 Entity lastResolved = EntityBad;
1097 auto resolve_cached = [&](Entity e) {
1098 if (e == lastKey)
1099 return lastResolved;
1100 lastKey = e;
1101 return lastResolved = resolve(e);
1102 };
1103
1104 cnt::darray_ext<Entity, 16> deleteTargets;
1105 {
1106 GAIA_PROF_SCOPE(cmdbuf::merges);
1107 for (uint32_t p = 0; p < m_ops.size();) {
1108 GAIA_PROF_SCOPE(cmdbuf::merge);
1109
1110 const Entity tgtKey = m_ops[p].target;
1111
1112 const bool tgtIsTemp = is_tmp(tgtKey);
1113 const uint32_t ti = tgtIsTemp ? tgtKey.id() : 0u;
1114 const Entity tgtReal =
1115 tgtIsTemp ? (ti < m_temp2real.size() ? m_temp2real[ti] : EntityBad) : resolve_cached(tgtKey);
1116
1117 // Range for this target
1118 uint32_t q = p;
1119 bool hasDelEntity = false;
1120 while (q < m_ops.size() && m_ops[q].target == tgtKey) {
1121 if (m_ops[q].type == OpType::DEL_ENTITY)
1122 hasDelEntity = true;
1123 ++q;
1124 }
1125
1126 // Skip canceled or non-existent temporary entities
1127 if (tgtReal == EntityBad) {
1128 p = q;
1129 continue;
1130 }
1131 if (tgtIsTemp && is_canceled_temp(ti)) {
1132 p = q;
1133 continue;
1134 }
1135
1136 if GAIA_UNLIKELY (m_ops[p].other != m_ops[q - 1].other &&
1137 (m_ops[p].type != OpType::SET_COMPONENT || m_ops[q - 1].type != OpType::SET_COMPONENT) &&
1138 replay_batch(p, q, tgtReal) && p == q) {
1139 if (hasDelEntity)
1140 deleteTargets.push_back(tgtReal);
1141 p = q;
1142 continue;
1143 }
1144
1145 // Emit relation groups.
1146 // Inside [p..q) range (same target), process groups by 'other'.
1147 // We perform per-component reduction.
1148 for (uint32_t i = p; i < q;) {
1149 const Entity othKey = m_ops[i].other;
1150 const Entity pairTargetKey = m_ops[i].pairTarget;
1151 const Entity othReal = resolve_object(m_ops[i]);
1152
1153 // Group ops with same (target, other)
1154 uint32_t j = i + 1;
1155 while (j < q && m_ops[j].other == othKey && m_ops[j].pairTarget == pairTargetKey)
1156 ++j;
1157
1158 if (tgtReal != EntityBad) {
1159 const uint32_t groupSize = j - i;
1160 // Fast path - single op
1161 if (groupSize == 1) {
1162 replay_op(m_ops[i], tgtReal, othReal);
1163 }
1164 // Slow path: merge multiple ops
1165 else {
1166 uint8_t mask = 0;
1167 uint32_t dataPos = 0;
1168
1169 for (uint32_t k = i; k < j; ++k) {
1170 const Op& op = m_ops[k];
1171 switch (op.type) {
1172 case OpType::ADD_COMPONENT:
1173 mask |= F_ADD;
1174 break;
1175 case OpType::ADD_COMPONENT_DATA:
1176 mask |= F_ADD_DATA;
1177 dataPos = op.off;
1178 break;
1179 case OpType::SET_COMPONENT:
1180 mask |= F_SET;
1181 dataPos = op.off;
1182 break;
1183 case OpType::DEL_COMPONENT:
1184 // A delete discards payloads from the preceding component lifetime.
1185 mask = F_DEL;
1186 break;
1187 default:
1188 break;
1189 }
1190 }
1191
1192 const bool hasAdd = mask & F_ADD;
1193 const bool hasAddData = mask & F_ADD_DATA;
1194 const bool hasSet = mask & F_SET;
1195 const bool hasDel = mask & F_DEL;
1196
1197 // 1) DEL present: reduce the surviving component lifetime.
1198 // Replay in order when World policies or observable side effects require it.
1199 if (hasDel) {
1200 replay_merged(i, j, tgtReal, othReal, mask, dataPos);
1201 }
1202 // 2) ADD_WITH_DATA or ADD+SET = ADD_WITH_DATA
1203 else if (hasAddData || (hasAdd && hasSet)) {
1204 replay_add_data(tgtReal, othReal, dataPos);
1205 }
1206 // 3) ADD only
1207 else if (hasAdd) {
1208 replay_add(tgtReal, othReal);
1209 }
1210 // 4) SET only
1211 else if (hasSet) {
1212 replay_data(tgtReal, othReal, dataPos);
1213 }
1214 }
1215 }
1216
1217 // Advance to next component group
1218 i = j;
1219 }
1220
1221 // Apply deletions after every relationship operation so sorting target groups cannot
1222 // invalidate an endpoint that a later group still needs to resolve.
1223 if (hasDelEntity)
1224 deleteTargets.push_back(tgtReal);
1225
1226 // Advance to next target group
1227 p = q;
1228 }
1229 }
1230 for (auto entity: deleteTargets) {
1231 if (m_world.valid(entity))
1232 m_world.del(entity);
1233 }
1234
1235 clear();
1236 }
1237 };
