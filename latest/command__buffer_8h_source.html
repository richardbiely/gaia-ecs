282 GAIA_NODISCARD Entity resolve_object(const Op& op) const {
283 if (op.pairTarget == EntityBad)
284 return resolve(op.other);
285
286 const auto relation = resolve(op.other);
287 const auto target = resolve(op.pairTarget);
288 if (relation == EntityBad || target == EntityBad)
289 return EntityBad;
290
291 return (Entity)Pair(relation, target);
292 }

297 void replay_add(Entity target, Entity object) {
298 if (object.pair()) {
299 World::EntityBuilder(m_world, target).add(Pair(m_world.get(object.id()), m_world.get(object.gen())));
300 return;
301 }
302
303 const auto* pItem = m_world.comp_cache().find(object);
304 if (pItem != nullptr && pItem->comp.storage_type() == DataStorageType::Sparse) {

306 GAIA_ASSERT(mode != World::SparseStorageMode::None);
307 auto& store = m_world.sparse_component_store_erased_mut(object, *pItem);
308 (void)store.func_add(store.pStore, target);
309 m_world.finish_sparse_component_add_inter(target, object, mode);
310 return;

331 void replay_data(Entity target, Entity object, uint32_t dataPos, bool finishWrite = true) {
332 auto serializer = ser::make_serializer(m_data);
333 serializer.seek(dataPos);
334 const auto* pItem =
335 object.pair() ? m_world.comp_cache().find_pair_payload(object) : m_world.comp_cache().find(object);
336 GAIA_ASSERT(pItem != nullptr);
337 if (pItem == nullptr)
338 return;
339 const auto& item = *pItem;
340
341 if (m_world.component_uses_sparse_storage(object)) {
342 const auto payload = m_world.mut_raw(target, object);
343 GAIA_ASSERT(payload.valid());
344 if (payload.valid())
345 item.load(serializer, payload.data, 0, 1, 1);
346 if (finishWrite)
347 m_world.finish_write(target, object);
348 return;
349 }
350
351 const auto& ec = target.pair() ? m_world.fetch(target) : m_world.m_recs.entities[target.id()];
352 const auto row = target.kind() == EntityKind::EK_Uni ? 0U : ec.row;
353 const auto compIdx = ec.pChunk->comp_idx(object);
354 auto* pComponentData = (void*)ec.pChunk->comp_ptr_mut(compIdx, 0);
355 item.load(serializer, pComponentData, row, row + 1, ec.pChunk->capacity());
356 if (finishWrite)
357 m_world.finish_write(target, object);
358 }

400 GAIA_NODISCARD bool is_canceled_temp(uint32_t idx) const {
401 const uint32_t base = idx * 3;
402 if (base + 2 >= m_tmpFlags.size())
403 return false;
404
405 const bool destroy = m_tmpFlags.test(base);
406 const bool usedOther = m_tmpFlags.test(base + 2);
407
408 // If deleted in this batch and not referenced by others, cancel entirely
409 return destroy && !usedOther;
410 }

413 GAIA_NODISCARD Entity add_temp(EntityKind kind) {
414 m_temp2real.push_back(EntityBad);
415 Entity tmp(m_nextTemp++, 0, true, false, kind);
416 // Use the unused flag to mark temporary entities
417 tmp.data.tmp = 1;
418 return tmp;
419 }

425 GAIA_NODISCARD static bool less_op_key(const Op& a, const Op& b) {
426 if (a.target != b.target)
427 return a.target < b.target;
428
429 const bool aIsPair = a.pairTarget != EntityBad;
430 const bool bIsPair = b.pairTarget != EntityBad;
431 if (aIsPair != bIsPair)
432 return !aIsPair;
433 if (aIsPair)
434 return false;
435
436 if (a.other != b.other)
437 return a.other < b.other;
438 return false;
439 }

443 void check_sort(const Op& op) {
444 if (!m_ops.empty() && less_op_key(op, m_ops.back()))
445 m_needsSort = true;

467 public:
469 void commit() {
470 core::lock_scope lock(m_acc);
471
472 if (m_ops.empty())
473 return;
474
475 GAIA_PROF_SCOPE(cmdbuf::commit);
476
477 // Build flags + allocate entities
478 if (m_nextTemp > 0) {
479 GAIA_PROF_SCOPE(cmdbuf::alloc);
480
481 // Bit layout for each temporary entity:
482 // bit 0 -> marked for destruction (DEL_ENTITY recorded)
483 // bit 1 -> used as a relation target (appeared as target in component ops)
484 // bit 2 -> used as a relation source / dependency (appeared as other in component ops)
485 m_tmpFlags.resize(m_nextTemp * 3);
486
487 // Pre-map surviving temps for ADD/CPY (avoid allocating canceled temps)
488 if (m_temp2real.size() < m_nextTemp) {
489 const auto from = m_temp2real.size();
490 m_temp2real.resize(m_nextTemp);
491 GAIA_FOR2(from, m_nextTemp) m_temp2real[i] = EntityBad;
492 }
493
494 // Build all flags
495 for (const Op& o: m_ops) {
496 // Set flag bits
497 if (is_tmp(o.target)) {
498 const uint32_t ti = o.target.id();
499
500 if (o.type == OpType::DEL_ENTITY)
501 m_tmpFlags.set((ti * 3) + 0, true);
502 else if (is_rel(o.type))
503 m_tmpFlags.set((ti * 3) + 1, true);
504 }
505
506 if (is_tmp(o.other) && o.other.id() < m_tmpFlags.size())
507 m_tmpFlags.set((o.other.id() * 3) + 2, true);
508 if (is_tmp(o.pairTarget) && o.pairTarget.id() < m_tmpFlags.size())
509 m_tmpFlags.set((o.pairTarget.id() * 3) + 2, true);
510 }
511
512 // Allocate real entities for the surviving temporaries
513 for (const Op& o: m_ops) {
514 if (!is_tmp(o.target))
515 continue;
516
517 const uint32_t ti = o.target.id();
518 if (is_canceled_temp(ti))
519 continue;
520
521 if (o.type == OpType::ADD_ENTITY) {
522 if (m_temp2real[ti] == EntityBad)
523 m_temp2real[ti] = m_world.add(o.target.kind());
524 } else if (o.type == OpType::CPY_ENTITY) {
525 if (m_temp2real[ti] == EntityBad) {
526 const Entity src = resolve(o.other);
527 if (src != EntityBad)
528 m_temp2real[ti] = m_world.copy(src);
529 }
530 }
531 }
532 }
533
534 // Sort by (target, other), reduce last-wins, apply relations.
535 // DEL_COMPONENT last per target.
536 if (m_needsSort) {
537 GAIA_PROF_SCOPE(cmdbuf::sort);
538
539 m_needsSort = false;
540 core::sort(m_ops.begin(), m_ops.end(), [](const Op& a, const Op& b) {
541 if (less_op_key(a, b))
542 return true;
543 if (less_op_key(b, a))
544 return false;
545 return a.order < b.order;
546 });
547 }
548
549 // Replay batched operations
550 Entity lastKey = EntityBad;
551 Entity lastResolved = EntityBad;
552 auto resolve_cached = [&](Entity e) {
553 if (e == lastKey)
554 return lastResolved;
555 lastKey = e;
556 return lastResolved = resolve(e);
557 };
558
559 cnt::darray_ext<Entity, 16> deleteTargets;
560 {
561 GAIA_PROF_SCOPE(cmdbuf::merges);
562 for (uint32_t p = 0; p < m_ops.size();) {
563 GAIA_PROF_SCOPE(cmdbuf::merge);
564
565 const Entity tgtKey = m_ops[p].target;
566
567 const bool tgtIsTemp = is_tmp(tgtKey);
568 const uint32_t ti = tgtIsTemp ? tgtKey.id() : 0u;
569 const Entity tgtReal =
570 tgtIsTemp ? (ti < m_temp2real.size() ? m_temp2real[ti] : EntityBad) : resolve_cached(tgtKey);
571
572 // Range for this target
573 uint32_t q = p;
574 bool hasDelEntity = false;
575 while (q < m_ops.size() && m_ops[q].target == tgtKey) {
576 if (m_ops[q].type == OpType::DEL_ENTITY)
577 hasDelEntity = true;
578 ++q;
579 }
580
581 // Skip canceled or non-existent temporary entities
582 if (tgtReal == EntityBad) {
583 p = q;
584 continue;
585 }
586 if (tgtIsTemp && is_canceled_temp(ti)) {
587 p = q;
588 continue;
589 }
590
591 enum : uint8_t { F_ADD = 1 << 0, F_ADD_DATA = 1 << 1, F_SET = 1 << 2, F_DEL = 1 << 3 };
592
593 // Emit relation groups.
594 // Inside [p..q) range (same target), process groups by 'other'.
595 // We perform per-component reduction.
596 for (uint32_t i = p; i < q;) {
597 const Entity othKey = m_ops[i].other;
598 const Entity pairTargetKey = m_ops[i].pairTarget;
599 const Entity othReal = resolve_object(m_ops[i]);
600
601 // Group ops with same (target, other)
602 uint32_t j = i + 1;
603 while (j < q && m_ops[j].other == othKey && m_ops[j].pairTarget == pairTargetKey)
604 ++j;
605
606 if (tgtReal != EntityBad) {
607 const uint32_t groupSize = j - i;
608 // Fast path - single op
609 if (groupSize == 1) {
610 const Op& op = m_ops[i];
611 switch (op.type) {
612 case OpType::DEL_COMPONENT:
613 replay_del(tgtReal, othReal);
614 break;
615 case OpType::ADD_COMPONENT:
616 replay_add(tgtReal, othReal);
617 break;
618 case OpType::ADD_COMPONENT_DATA:
619 replay_add_data(tgtReal, othReal, op.off);
620 break;
621 case OpType::SET_COMPONENT:
622 replay_data(tgtReal, othReal, op.off);
623 break;
624 default:
625 break;
626 }
627 }
628 // Slow path: merge multiple ops
629 else {
630 uint8_t mask = 0;
631 uint32_t dataPos = 0;
632
633 for (uint32_t k = i; k < j; ++k) {
634 const Op& op = m_ops[k];
635 switch (op.type) {
636 case OpType::ADD_COMPONENT:
637 mask |= F_ADD;
638 break;
639 case OpType::ADD_COMPONENT_DATA:
640 mask |= F_ADD_DATA;
641 dataPos = op.off;
642 break;
643 case OpType::SET_COMPONENT:
644 mask |= F_SET;
645 dataPos = op.off;
646 break;
647 case OpType::DEL_COMPONENT:
648 mask |= F_DEL;
649 break;
650 default:
651 break;
652 }
653 }
654
655 const bool hasAdd = mask & F_ADD;
656 const bool hasAddData = mask & F_ADD_DATA;
657 const bool hasSet = mask & F_SET;
658 const bool hasDel = mask & F_DEL;
659
660 // 1) ADD(+DATA) + DEL = no-op
661 if (hasDel && (hasAdd || hasAddData)) {
662 }
663 // 2) DEL only
664 else if (hasDel) {
665 replay_del(tgtReal, othReal);
666 }
667 // 3) ADD_WITH_DATA or ADD+SET = ADD_WITH_DATA
668 else if (hasAddData || (hasAdd && hasSet)) {
669 replay_add_data(tgtReal, othReal, dataPos);
670 }
671 // 4) ADD only
672 else if (hasAdd) {
673 replay_add(tgtReal, othReal);
674 }
675 // 5) SET only
676 else if (hasSet) {
677 replay_data(tgtReal, othReal, dataPos);
678 }
679 }
680 }
681
682 // Advance to next component group
683 i = j;
684 }
685
686 // Apply deletions after every relationship operation so sorting target groups cannot
687 // invalidate an endpoint that a later group still needs to resolve.
688 if (hasDelEntity)
689 deleteTargets.push_back(tgtReal);
690
691 // Advance to next target group
692 p = q;
693 }
694 }
695 for (auto entity: deleteTargets) {
696 if (m_world.valid(entity))
697 m_world.del(entity);
698 }
699
700 clear();
701 }
702 };
