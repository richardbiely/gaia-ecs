475 GAIA_NODISCARD Entity resolve_object(const Op& op) const {
476 if (op.pairTarget == EntityBad)
477 return resolve(op.other);
478
479 const auto relation = resolve(op.other);
480 const auto target = resolve(op.pairTarget);
481 if (relation == EntityBad || target == EntityBad)
482 return EntityBad;
483
484 return (Entity)Pair(relation, target);
485 }

491 void allocate_instantiate_group(uint32_t firstIdx) {
492 const Op& first = m_ops[firstIdx];
493 const Entity prefab = resolve(first.other);
494 if (prefab == EntityBad)
495 return;
496
497 Entity parent = EntityBad;
498 if (first.pairTarget != EntityBad) {
499 parent = resolve(first.pairTarget);
500 if (parent == EntityBad)
501 return;
502 }
503
504 cnt::darray_ext<uint32_t, 16> tempIds;
505 for (uint32_t j = firstIdx; j < m_ops.size(); ++j) {
506 const Op& op = m_ops[j];
507 if (op.type != OpType::INSTANTIATE_ENTITY || op.other != first.other || op.pairTarget != first.pairTarget)
508 continue;
509 if (!is_tmp(op.target))
510 continue;
511
512 const uint32_t ti = op.target.id();
513 if (ti >= m_temp2real.size() || m_temp2real[ti] != EntityBad || is_canceled_temp(ti))
514 continue;
515
516 tempIds.push_back(ti);
517 }
518
519 if (tempIds.empty())
520 return;
521
522 if (tempIds.size() == 1) {
523 m_temp2real[tempIds[0]] =
524 parent == EntityBad ? m_world.instantiate(prefab) : m_world.instantiate(prefab, parent);
525 return;
526 }
527
528 uint32_t mapped = 0;
529 auto mapRoot = [&](Entity instance) {
530 GAIA_ASSERT(mapped < tempIds.size());
531 m_temp2real[tempIds[mapped++]] = instance;
532 };
533 if (parent == EntityBad)
534 m_world.instantiate_n(prefab, tempIds.size(), mapRoot);
535 else
536 m_world.instantiate_n(prefab, parent, tempIds.size(), mapRoot);
537 GAIA_ASSERT(mapped == tempIds.size());
538 }

577 void replay_data(Entity target, Entity object, uint32_t dataPos, bool finishWrite = true) {
578 auto serializer = ser::make_serializer(m_data);
579 serializer.seek(dataPos);
580 const auto* pItem =
581 object.pair() ? m_world.comp_cache().find_pair_payload(object) : m_world.comp_cache().find(object);
582 GAIA_ASSERT(pItem != nullptr);
583 if (pItem == nullptr)
584 return;
585 const auto& item = *pItem;
586
587 if (m_world.component_uses_sparse_storage(object)) {
588 const auto payload = m_world.mut_raw(target, object);
589 GAIA_ASSERT(payload.valid());
590 if (payload.valid())
591 item.load(serializer, payload.data, 0, 1, 1);
592 if (finishWrite)
593 m_world.finish_write(target, object);
594 return;
595 }
596
597 const auto& ec = target.pair() ? m_world.fetch(target) : m_world.m_recs.entities[target.id()];
598 const auto row = ec.row;
599 const auto compIdx = ec.pChunk->comp_idx(object);
600 auto* pComponentData = (void*)ec.pChunk->comp_ptr_mut(compIdx, 0);
601 item.load(serializer, pComponentData, row, row + 1, ec.pChunk->capacity());
602 if (finishWrite)
603 m_world.finish_write(target, object);
604 }

648 GAIA_NODISCARD bool is_canceled_temp(uint32_t idx) const {
649 const uint32_t base = idx * 3;
650 if (base + 2 >= m_tmpFlags.size())
651 return false;
652
653 const bool destroy = m_tmpFlags.test(base);
654 const bool usedOther = m_tmpFlags.test(base + 2);
655
656 // If deleted in this batch and not referenced by others, cancel entirely
657 return destroy && !usedOther;
658 }

662 GAIA_NODISCARD Entity add_temp() {
663 m_temp2real.push_back(EntityBad);
664 Entity tmp(m_nextTemp++, 0, true, false);
665 // Use the unused flag to mark temporary entities
666 tmp.data.tmp = 1;
667 return tmp;
668 }

674 GAIA_NODISCARD static bool less_op_key(const Op& a, const Op& b) {
675 if (a.target != b.target)
676 return a.target < b.target;
677
678 const bool aIsPair = a.pairTarget != EntityBad;
679 const bool bIsPair = b.pairTarget != EntityBad;
680 if (aIsPair != bIsPair)
681 return !aIsPair;
682 if (aIsPair)
683 return false;
684
685 if (a.other != b.other)
686 return a.other < b.other;
687 return false;
688 }

717 public:
721 void commit() {
722 core::lock_scope lock(m_acc);
723
724 if (m_ops.empty())
725 return;
726
727 GAIA_PROF_SCOPE(cmdbuf::commit);
728
729 // Build flags + allocate entities
730 if (m_nextTemp > 0) {
731 GAIA_PROF_SCOPE(cmdbuf::alloc);
732
733 // Bit layout for each temporary entity:
734 // bit 0 -> marked for destruction (DEL_ENTITY recorded)
735 // bit 1 -> used as a relation target (appeared as target in component ops)
736 // bit 2 -> used as a relation source / dependency (appeared as other in component ops)
737 m_tmpFlags.resize(m_nextTemp * 3);
738
739 // Pre-map surviving temps for ADD/CPY/INSTANTIATE (avoid allocating canceled temps)
740 if (m_temp2real.size() < m_nextTemp) {
741 const auto from = m_temp2real.size();
742 m_temp2real.resize(m_nextTemp);
743 GAIA_FOR2(from, m_nextTemp) m_temp2real[i] = EntityBad;
744 }
745
746 // Build all flags
747 for (const Op& o: m_ops) {
748 // Set flag bits
749 if (is_tmp(o.target)) {
750 const uint32_t ti = o.target.id();
751
752 if (o.type == OpType::DEL_ENTITY)
753 m_tmpFlags.set((ti * 3) + 0, true);
754 else if (is_rel(o.type))
755 m_tmpFlags.set((ti * 3) + 1, true);
756 }
757
758 if (is_tmp(o.other) && o.other.id() < m_tmpFlags.size())
759 m_tmpFlags.set((o.other.id() * 3) + 2, true);
760 if (is_tmp(o.pairTarget) && o.pairTarget.id() < m_tmpFlags.size())
761 m_tmpFlags.set((o.pairTarget.id() * 3) + 2, true);
762 }
763
764 // Allocate real entities for the surviving temporaries
765 for (uint32_t i = 0; i < m_ops.size(); ++i) {
766 const Op& o = m_ops[i];
767 if (!is_tmp(o.target))
768 continue;
769
770 const uint32_t ti = o.target.id();
771 if (is_canceled_temp(ti) || m_temp2real[ti] != EntityBad)
772 continue;
773
774 if (o.type == OpType::ADD_ENTITY) {
775 m_temp2real[ti] = m_world.add();
776 } else if (o.type == OpType::CPY_ENTITY) {
777 const Entity src = resolve(o.other);
778 if (src != EntityBad)
779 m_temp2real[ti] = m_world.copy(src);
780 } else if (o.type == OpType::INSTANTIATE_ENTITY) {
781 allocate_instantiate_group(i);
782 }
783 }
784 }
785
786 // Sort by (target, other), reduce last-wins, apply relations.
787 // DEL_COMPONENT last per target.
788 if (m_needsSort) {
789 GAIA_PROF_SCOPE(cmdbuf::sort);
790
791 m_needsSort = false;
792 core::sort(m_ops.begin(), m_ops.end(), [](const Op& a, const Op& b) {
793 if (less_op_key(a, b))
794 return true;
795 if (less_op_key(b, a))
796 return false;
797 return a.order < b.order;
798 });
799 }
800
801 // Replay batched operations
802 Entity lastKey = EntityBad;
803 Entity lastResolved = EntityBad;
804 auto resolve_cached = [&](Entity e) {
805 if (e == lastKey)
806 return lastResolved;
807 lastKey = e;
808 return lastResolved = resolve(e);
809 };
810
811 cnt::darray_ext<Entity, 16> deleteTargets;
812 {
813 GAIA_PROF_SCOPE(cmdbuf::merges);
814 for (uint32_t p = 0; p < m_ops.size();) {
815 GAIA_PROF_SCOPE(cmdbuf::merge);
816
817 const Entity tgtKey = m_ops[p].target;
818
819 const bool tgtIsTemp = is_tmp(tgtKey);
820 const uint32_t ti = tgtIsTemp ? tgtKey.id() : 0u;
821 const Entity tgtReal =
822 tgtIsTemp ? (ti < m_temp2real.size() ? m_temp2real[ti] : EntityBad) : resolve_cached(tgtKey);
823
824 // Range for this target
825 uint32_t q = p;
826 bool hasDelEntity = false;
827 while (q < m_ops.size() && m_ops[q].target == tgtKey) {
828 if (m_ops[q].type == OpType::DEL_ENTITY)
829 hasDelEntity = true;
830 ++q;
831 }
832
833 // Skip canceled or non-existent temporary entities
834 if (tgtReal == EntityBad) {
835 p = q;
836 continue;
837 }
838 if (tgtIsTemp && is_canceled_temp(ti)) {
839 p = q;
840 continue;
841 }
842
843 enum : uint8_t { F_ADD = 1 << 0, F_ADD_DATA = 1 << 1, F_SET = 1 << 2, F_DEL = 1 << 3 };
844
845 // Emit relation groups.
846 // Inside [p..q) range (same target), process groups by 'other'.
847 // We perform per-component reduction.
848 for (uint32_t i = p; i < q;) {
849 const Entity othKey = m_ops[i].other;
850 const Entity pairTargetKey = m_ops[i].pairTarget;
851 const Entity othReal = resolve_object(m_ops[i]);
852
853 // Group ops with same (target, other)
854 uint32_t j = i + 1;
855 while (j < q && m_ops[j].other == othKey && m_ops[j].pairTarget == pairTargetKey)
856 ++j;
857
858 if (tgtReal != EntityBad) {
859 const uint32_t groupSize = j - i;
860 // Fast path - single op
861 if (groupSize == 1) {
862 const Op& op = m_ops[i];
863 switch (op.type) {
864 case OpType::DEL_COMPONENT:
865 replay_del(tgtReal, othReal);
866 break;
867 case OpType::ADD_COMPONENT:
868 replay_add(tgtReal, othReal);
869 break;
870 case OpType::ADD_COMPONENT_DATA:
871 replay_add_data(tgtReal, othReal, op.off);
872 break;
873 case OpType::SET_COMPONENT:
874 replay_data(tgtReal, othReal, op.off);
875 break;
876 default:
877 break;
878 }
879 }
880 // Slow path: merge multiple ops
881 else {
882 uint8_t mask = 0;
883 uint32_t dataPos = 0;
884
885 for (uint32_t k = i; k < j; ++k) {
886 const Op& op = m_ops[k];
887 switch (op.type) {
888 case OpType::ADD_COMPONENT:
889 mask |= F_ADD;
890 break;
891 case OpType::ADD_COMPONENT_DATA:
892 mask |= F_ADD_DATA;
893 dataPos = op.off;
894 break;
895 case OpType::SET_COMPONENT:
896 mask |= F_SET;
897 dataPos = op.off;
898 break;
899 case OpType::DEL_COMPONENT:
900 mask |= F_DEL;
901 break;
902 default:
903 break;
904 }
905 }
906
907 const bool hasAdd = mask & F_ADD;
908 const bool hasAddData = mask & F_ADD_DATA;
909 const bool hasSet = mask & F_SET;
910 const bool hasDel = mask & F_DEL;
911
912 // 1) ADD(+DATA) + DEL = no-op
913 if (hasDel && (hasAdd || hasAddData)) {
914 }
915 // 2) DEL only
916 else if (hasDel) {
917 replay_del(tgtReal, othReal);
918 }
919 // 3) ADD_WITH_DATA or ADD+SET = ADD_WITH_DATA
920 else if (hasAddData || (hasAdd && hasSet)) {
921 replay_add_data(tgtReal, othReal, dataPos);
922 }
923 // 4) ADD only
924 else if (hasAdd) {
925 replay_add(tgtReal, othReal);
926 }
927 // 5) SET only
928 else if (hasSet) {
929 replay_data(tgtReal, othReal, dataPos);
930 }
931 }
932 }
933
934 // Advance to next component group
935 i = j;
936 }
937
938 // Apply deletions after every relationship operation so sorting target groups cannot
939 // invalidate an endpoint that a later group still needs to resolve.
940 if (hasDelEntity)
941 deleteTargets.push_back(tgtReal);
942
943 // Advance to next target group
944 p = q;
945 }
946 }
947 for (auto entity: deleteTargets) {
948 if (m_world.valid(entity))
949 m_world.del(entity);
950 }
951
952 clear();
953 }
954 };
955 } // namespace detail
