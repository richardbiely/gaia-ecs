67 void ensure_source_capacity(Entity source) {
68 const auto required = (uint32_t)source.id() + 1;
69 if (srcToTgt.size() >= required)

156 void remove_target_source(Entity target, Entity source) {
157 GAIA_ASSERT(target.id() < tgtToSrc.size());
158 if (target.id() >= tgtToSrc.size())
159 return;
160
161 auto& sources = tgtToSrc[target.id()];
162 const auto idx = target_source_index(source);
163 GAIA_ASSERT(idx != BadIndex && idx < sources.size());
164 if (idx == BadIndex || idx >= sources.size())
165 return;
166
167 const auto lastIdx = (uint32_t)sources.size() - 1;
168 if (idx != lastIdx) {
169 const auto movedSource = sources[lastIdx];
170 sources[idx] = movedSource;
171 set_target_source_index(movedSource, idx);
172 }
173
174 sources.pop_back();
175 }

181 GAIA_NODISCARD bool set(Entity source, Entity target) {
182 if GAIA_LIKELY (!source.pair())
183 ensure_source_capacity(source);
184 const auto oldTarget = this->target(source);
185 if (oldTarget != EntityBad) {
186 if (oldTarget == target)
187 return false;
188
189 remove_target_source(oldTarget, source);
190 } else {
191 ++srcToTgtCnt;
192 }
193
194 ensure_target_capacity(target);
195 auto& sources = tgtToSrc[target.id()];
196 const auto sourceIdx = (uint32_t)sources.size();
197 if GAIA_UNLIKELY (source.pair())
198 pairSrcToTgt[EntityLookupKey(source)] = PairSourceRecord{target, sourceIdx};
199 else {
200 srcToTgt[source.id()] = target;
201 srcToTgtIdx[source.id()] = sourceIdx;
202 }
203 sources.push_back(source);
204
205 return true;
206 }

212 GAIA_NODISCARD bool remove(Entity source, Entity target) {
213 const auto oldTarget = this->target(source);
214 if (oldTarget == EntityBad)
215 return false;
216 if (target != EntityBad && oldTarget != target)
217 return false;
218
219 remove_target_source(oldTarget, source);
220 if GAIA_UNLIKELY (source.pair())
221 pairSrcToTgt.erase(EntityLookupKey(source));
222 else {
223 srcToTgt[source.id()] = EntityBad;
224 srcToTgtIdx[source.id()] = BadIndex;
225 }
226 GAIA_ASSERT(srcToTgtCnt > 0);
227 --srcToTgtCnt;
228 return true;
229 }
230
233 GAIA_NODISCARD bool empty() const {
234 return srcToTgtCnt == 0;
