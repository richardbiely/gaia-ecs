93 template <typename InputIt>
94 darr(InputIt first, InputIt last) {
95 const auto count = (size_type)core::distance(first, last);
96 resize(count);
97
98 if constexpr (std::is_pointer_v<InputIt>) {
99 for (size_type i = 0; i < count; ++i)
100 operator[](i) = first[i];
101 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
102 for (size_type i = 0; i < count; ++i)
103 operator[](i) = *(first[i]);
104 } else {
105 size_type i = 0;
106 for (auto it = first; it != last; ++it)
107 operator[](i++) = *it;
108 }
109 }

208 void reserve(size_type cap) {
209 if (cap <= m_cap)
210 return;
211
212 auto* pDataOld = m_pData;
213 m_pData = view_policy::template alloc<Allocator>(cap);
214 if (pDataOld != nullptr) {
215 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, cap, m_cnt);
216 mem::move_ctor_elements<T>(m_pData, pDataOld, m_cnt);
217 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
218 }
219
220 m_cap = cap;
221 }

225 void resize(size_type count) {
226 if (count == m_cnt)
227 return;
228
229 // Fresh allocation
230 if (m_pData == nullptr) {
231 if (count > 0) {
232 m_pData = view_policy::template alloc<Allocator>(count);
233 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, count, count);
234 core::call_ctor_n(data(), count);
235 m_cap = count;
236 m_cnt = count;
237 }
238 return;
239 }
240
241 // Resizing to a smaller size
242 if (count < m_cnt) {
243 // Destroy elements at the end
244 core::call_dtor_n(&data()[count], m_cnt - count);
245 GAIA_MEM_SANI_POP_N(value_size, m_pData, m_cap, m_cnt, m_cnt - count);
246
247 m_cnt = count;
248 return;
249 }
250
251 // Resizing to a bigger size but still within allocated capacity
252 if (count <= m_cap) {
253 // Construct new elements
254 GAIA_MEM_SANI_PUSH_N(value_size, m_pData, m_cap, m_cnt, count - m_cnt);
255 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
256
257 m_cnt = count;
258 return;
259 }
260
261 auto* pDataOld = m_pData;
262 m_pData = view_policy::template alloc<Allocator>(count);
263 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, count, count);
264 // Move old data to the new location
265 mem::move_ctor_elements<T>(m_pData, pDataOld, m_cnt);
266 // Default-construct new items
267 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
268 // Release old memory
269 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
270
271 m_cap = count;
272 m_cnt = count;
273 }

278 void resize(size_type count, const_reference value) {

281 return;
282 }

285 const value_type valueCopy = value;
286 resize(count);
287 for (size_type i = oldCount; i < m_cnt; ++i)
288 operator[](i) = valueCopy;
289 } else {
290 value_type valueCopy;
291 valueCopy = value;
292 resize(count);
293 for (size_type i = oldCount; i < m_cnt; ++i)
294 operator[](i) = valueCopy;
295 }

314 template <typename... Args>
315 decltype(auto) emplace_back(Args&&... args) {
316 if GAIA_UNLIKELY (m_cnt == m_cap) {
317 const auto cap = m_cap != 0 ? (m_cap * 3 + 1) / 2 : 4;
318 auto* pDataNew = view_policy::template alloc<Allocator>(cap);
319 GAIA_MEM_SANI_ADD_BLOCK(value_size, pDataNew, cap, m_cnt + 1);
321 core::call_ctor(&static_cast<T*>(static_cast<void*>(pDataNew))[m_cnt], GAIA_FWD(args)...);
322 mem::move_ctor_elements<T>(pDataNew, m_pData, m_cnt);
323 view_policy::template free<Allocator>(m_pData, m_cap, m_cnt);
324 m_pData = pDataNew;
325 m_cap = cap;
326 } else {
327 GAIA_MEM_SANI_PUSH(value_size, m_pData, m_cap, m_cnt);
328 core::call_ctor(&data()[m_cnt], GAIA_FWD(args)...);
329 }
330 return (reference)data()[m_cnt++];
331 }

334 void pop_back() noexcept {

336
337 auto* ptr = &data()[m_cnt - 1];
338 core::call_dtor(ptr);
339 GAIA_MEM_SANI_POP(value_size, m_pData, m_cap, m_cnt);
340
341 --m_cnt;
342 }

348 iterator insert(iterator pos, const T& arg) {
349 GAIA_ASSERT(pos >= data());
350 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
351
352 T value(arg);
353 const auto idxSrc = (size_type)core::distance(begin(), pos);
354 try_grow();
355 const auto idxDst = (size_type)core::distance(begin(), end());
356
357 GAIA_MEM_SANI_PUSH(value_size, m_pData, m_cap, m_cnt);
358 auto* ptr = &data()[idxSrc];
359 if (idxSrc == idxDst) {
360 core::call_ctor(ptr, std::move_if_noexcept(value));
361 } else {
362 mem::move_ctor_elements<T>((uint8_t*)&data()[idxDst], (uint8_t*)&data()[idxDst - 1], 1);
363 mem::shift_elements_right<T, false>(m_pData, idxDst - 1, idxSrc, m_cap);
364 mem::move_element<T, false>((uint8_t*)ptr, (uint8_t*)&value, 0, 0, 1, 1);
365 }
366
367 ++m_cnt;
368
369 return iterator(&data()[idxSrc]);
370 }

403 iterator erase(iterator pos) noexcept {
404 GAIA_ASSERT(pos >= data());
405 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
406
407 if (empty())
408 return end();
409
410 const auto idxSrc = (size_type)core::distance(begin(), pos);
411 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
412
413 mem::shift_elements_left<T, false>(m_pData, idxDst, idxSrc, m_cap);
414 // Destroy if it's the last element
415 auto* ptr = &data()[m_cnt - 1];
416 core::call_dtor(ptr);
417 GAIA_MEM_SANI_POP(value_size, m_pData, m_cap, m_cnt);
418
419 --m_cnt;
420
421 return iterator(&data()[idxSrc]);
422 }

428 iterator erase(iterator first, iterator last) noexcept {
429 GAIA_ASSERT(first >= data())
430 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
431 GAIA_ASSERT(last > first);
432 GAIA_ASSERT(last <= iterator(data() + size()));
433
434 if (empty())
435 return end();
436
437 const auto idxSrc = (size_type)core::distance(begin(), first);
438 const auto idxDst = size();
439 const auto cnt = (size_type)(last - first);
440
441 mem::shift_elements_left_fast<T, false>(m_pData, idxDst, idxSrc, cnt, m_cap);
442 // Destroy if it's the last element
443 auto* ptr = &data()[m_cnt - cnt];
444 core::call_dtor_n(ptr, cnt);
445 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, cnt);
446
447 m_cnt -= cnt;
448
449 return iterator(&data()[idxSrc]);
450 }

458 void shrink_to_fit() {
459 const auto cap = capacity();
460 const auto cnt = size();
461 if (cap == cnt)
462 return;
463
464 auto* pDataOld = m_pData;
465 m_pData = cnt != 0 ? view_policy::template alloc<Allocator>(cnt) : nullptr;
466 if (cnt != 0) {
467 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, cnt, cnt);
468 mem::move_ctor_elements<T>(m_pData, pDataOld, cnt);
469 }
470 view_policy::template free<Allocator>(pDataOld, cap, cnt);
471 m_cap = cnt;
