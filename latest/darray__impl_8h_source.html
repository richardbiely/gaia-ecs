65 void try_grow() {
66 const auto cnt = size();
67 const auto cap = capacity();
68
69 // Unless we reached the capacity don't do anything
70 if GAIA_LIKELY (cap != 0 && cnt < cap)
71 return;
72
73 // If no data is allocated go with at least 4 elements
74 if GAIA_UNLIKELY (m_pData == nullptr) {
75 m_pData = view_policy::template alloc<Allocator>(m_cap = 4);
76 return;
77 }
78
79 // We increase the capacity in multiples of 1.5 which is about the golden ratio (1.618).
80 // This effectively means we prefer more frequent allocations over memory fragmentation.
81 m_cap = (cap * 3 + 1) / 2;
82
83 auto* pDataOld = m_pData;
84 m_pData = view_policy::template alloc<Allocator>(m_cap);
85 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, m_cap, cnt);
86 mem::move_elements<T, false>(m_pData, pDataOld, cnt, 0, m_cap, cap);
87 view_policy::template free<Allocator>(pDataOld, cap, cnt);
88 }

112 template <typename InputIt>
113 darr(InputIt first, InputIt last) {
114 const auto count = (size_type)core::distance(first, last);
115 resize(count);
116
117 if constexpr (std::is_pointer_v<InputIt>) {
118 for (size_type i = 0; i < count; ++i)
119 operator[](i) = first[i];
120 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
121 for (size_type i = 0; i < count; ++i)
122 operator[](i) = *(first[i]);
123 } else {
124 size_type i = 0;
125 for (auto it = first; it != last; ++it)
126 operator[](i++) = *it;
127 }
128 }

227 void reserve(size_type cap) {
228 if (cap <= m_cap)
229 return;
230
231 auto* pDataOld = m_pData;
232 m_pData = view_policy::template alloc<Allocator>(cap);
233 if (pDataOld != nullptr) {
234 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, cap, m_cnt);
235 mem::move_elements<T, false>(m_pData, pDataOld, m_cnt, 0, cap, m_cap);
236 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
237 }
238

240 }

244 void resize(size_type count) {
245 if (count == m_cnt)
246 return;
247
248 // Fresh allocation
249 if (m_pData == nullptr) {
250 if (count > 0) {
251 m_pData = view_policy::template alloc<Allocator>(count);
252 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, count, count);
253 core::call_ctor_n(m_pData, count);
254 m_cap = count;
255 m_cnt = count;
256 }
257 return;
258 }
259
260 // Resizing to a smaller size
261 if (count < m_cnt) {
262 // Destroy elements at the end
263 core::call_dtor_n(&data()[count], m_cnt - count);
264 GAIA_MEM_SANI_POP_N(value_size, m_pData, m_cap, m_cnt, m_cnt - count);
265
266 m_cnt = count;
267 return;
268 }
269
270 // Resizing to a bigger size but still within allocated capacity
271 if (count <= m_cap) {
272 // Construct new elements
273 GAIA_MEM_SANI_PUSH_N(value_size, m_pData, m_cap, m_cnt, count - m_cnt);
274 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
275
276 m_cnt = count;
277 return;
278 }
279
280 auto* pDataOld = m_pData;
281 m_pData = view_policy::template alloc<Allocator>(count);
282 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, count, count);
283 // Move old data to the new location
284 mem::move_elements<T, false>(m_pData, pDataOld, m_cnt, 0, count, m_cap);
285 // Default-construct new items
286 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
287 // Release old memory
288 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
289
290 m_cap = count;
291 m_cnt = count;
292 }

403 iterator erase(iterator pos) noexcept {
404 GAIA_ASSERT(pos >= data());
405 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
406
407 if (empty())
408 return end();
409
410 const auto idxSrc = (size_type)core::distance(begin(), pos);
411 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
412
413 mem::shift_elements_left<T, false>(m_pData, idxDst, idxSrc, m_cap);
414 // Destroy if it's the last element
415 auto* ptr = &data()[m_cnt - 1];
416 core::call_dtor(ptr);
417 GAIA_MEM_SANI_POP(value_size, m_pData, m_cap, m_cnt);
418
419 --m_cnt;
420
421 return iterator(&data()[idxSrc]);
422 }

428 iterator erase(iterator first, iterator last) noexcept {
429 GAIA_ASSERT(first >= data())
430 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
431 GAIA_ASSERT(last > first);
432 GAIA_ASSERT(last <= iterator(data() + size()));
433
434 if (empty())
435 return end();
436
437 const auto idxSrc = (size_type)core::distance(begin(), first);
438 const auto idxDst = size();
439 const auto cnt = (size_type)(last - first);
440
441 mem::shift_elements_left_fast<T, false>(m_pData, idxDst, idxSrc, cnt, m_cap);
442 // Destroy if it's the last element
443 auto* ptr = &data()[m_cnt - cnt];
444 core::call_dtor_n(ptr, cnt);
445 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, cnt);
446
447 m_cnt -= cnt;
448
449 return iterator(&data()[idxSrc]);
450 }

477 template <typename Func>
478 auto retain(Func&& func) {
479 size_type erased = 0;
480 size_type idxDst = 0;
481 size_type idxSrc = 0;
482
483 while (idxSrc < m_cnt) {
484 if (func(operator[](idxSrc))) {
485 if (idxDst < idxSrc) {
486 mem::move_element<T, false>(m_pData, m_pData, idxDst, idxSrc, m_cap, m_cap);
487 auto* ptr = &data()[idxSrc];
488 core::call_dtor(ptr);
489 }
490 ++idxDst;
491 } else {
492 auto* ptr = &data()[idxSrc];
493 core::call_dtor(ptr);
494 ++erased;
495 }
496
497 ++idxSrc;
498 }
499
500 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, erased);
501
502 m_cnt -= erased;
503 return idxDst;
504 }
