93 template <typename InputIt>
94 darr(InputIt first, InputIt last) {
95 const auto count = (size_type)core::distance(first, last);
96 resize(count);
97
98 if constexpr (std::is_pointer_v<InputIt>) {
99 for (size_type i = 0; i < count; ++i)
100 operator[](i) = first[i];
101 } else if constexpr (std::is_same_v<typename InputIt::iterator_category, core::random_access_iterator_tag>) {
102 for (size_type i = 0; i < count; ++i)
103 operator[](i) = *(first[i]);
104 } else {
105 size_type i = 0;
106 for (auto it = first; it != last; ++it)
107 operator[](i++) = *it;
108 }
109 }

208 void reserve(size_type cap) {
209 if (cap <= m_cap)
210 return;
211
212 auto* pDataOld = m_pData;
213 m_pData = view_policy::template alloc<Allocator>(cap);
214 if (pDataOld != nullptr) {

216 mem::move_ctor_elements<T>(m_pData, pDataOld, m_cnt);
217 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
218 }

220 m_cap = cap;

226 void resize(size_type count, ResizePolicy policy = ResizePolicy::Exact) {
227 if (count == m_cnt)
228 return;
229
230 // Fresh allocation
231 if (m_pData == nullptr) {
232 if (count > 0) {
233 m_pData = view_policy::template alloc<Allocator>(count);
234 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, count, count);
235 core::call_ctor_n(data(), count);
236 m_cap = count;
237 m_cnt = count;
238 }
239 return;
240 }
241
242 // Resizing to a smaller size
243 if (count < m_cnt) {
244 // Destroy elements at the end
245 core::call_dtor_n(&data()[count], m_cnt - count);
246 GAIA_MEM_SANI_POP_N(value_size, m_pData, m_cap, m_cnt, m_cnt - count);
247
248 m_cnt = count;
249 return;
250 }
251
252 // Resizing to a bigger size but still within allocated capacity
253 if (count <= m_cap) {
254 // Construct new elements
255 GAIA_MEM_SANI_PUSH_N(value_size, m_pData, m_cap, m_cnt, count - m_cnt);
256 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
257
258 m_cnt = count;
259 return;
260 }
261
262 const auto cap = detail::resize_capacity(policy, m_cap, count);
263 auto* pDataOld = m_pData;
264 m_pData = view_policy::template alloc<Allocator>(cap);
265 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, cap, count);
266 // Move old data to the new location
267 mem::move_ctor_elements<T>(m_pData, pDataOld, m_cnt);
268 // Default-construct new items
269 core::call_ctor_n(&data()[m_cnt], count - m_cnt);
270 // Release old memory
271 view_policy::template free<Allocator>(pDataOld, m_cap, m_cnt);
272
273 m_cap = cap;
274 m_cnt = count;
275 }

281 void resize(size_type count, const_reference value, ResizePolicy policy = ResizePolicy::Exact) {
282 if (count <= m_cnt) {
283 resize(count, policy);
284 return;
285 }
286 const auto oldCount = m_cnt;
287 if constexpr (std::is_copy_constructible_v<value_type>) {
288 const value_type valueCopy = value;
289 resize(count, policy);
290 for (size_type i = oldCount; i < m_cnt; ++i)
291 operator[](i) = valueCopy;
292 } else {
293 value_type valueCopy;
294 valueCopy = value;
295 resize(count, policy);
296 for (size_type i = oldCount; i < m_cnt; ++i)
297 operator[](i) = valueCopy;
298 }

317 template <typename... Args>
318 decltype(auto) emplace_back(Args&&... args) {
319 if GAIA_UNLIKELY (m_cnt == m_cap) {
320 const auto cap = m_cap != 0 ? (m_cap * 3 + 1) / 2 : 4;
321 auto* pDataNew = view_policy::template alloc<Allocator>(cap);
322 GAIA_MEM_SANI_ADD_BLOCK(value_size, pDataNew, cap, m_cnt + 1);
324 core::call_ctor(&static_cast<T*>(static_cast<void*>(pDataNew))[m_cnt], GAIA_FWD(args)...);
325 mem::move_ctor_elements<T>(pDataNew, m_pData, m_cnt);
326 view_policy::template free<Allocator>(m_pData, m_cap, m_cnt);
327 m_pData = pDataNew;
328 m_cap = cap;
329 } else {
330 GAIA_MEM_SANI_PUSH(value_size, m_pData, m_cap, m_cnt);
331 core::call_ctor(&data()[m_cnt], GAIA_FWD(args)...);
332 }
333 return (reference)data()[m_cnt++];
334 }

341 core::call_dtor(ptr);
342 GAIA_MEM_SANI_POP(value_size, m_pData, m_cap, m_cnt);

345 }

351 iterator insert(iterator pos, const T& arg) {
352 GAIA_ASSERT(pos >= data());
353 GAIA_ASSERT(pos <= iterator(data() + size()));
354
355 T value(arg);
356 const auto idxSrc = (size_type)core::distance(begin(), pos);
357 try_grow();
358 const auto idxDst = (size_type)core::distance(begin(), end());
359
360 GAIA_MEM_SANI_PUSH(value_size, m_pData, m_cap, m_cnt);
361 auto* ptr = &data()[idxSrc];
362 if (idxSrc == idxDst) {
363 core::call_ctor(ptr, std::move_if_noexcept(value));
364 } else {
365 mem::move_ctor_elements<T>((uint8_t*)&data()[idxDst], (uint8_t*)&data()[idxDst - 1], 1);
366 mem::shift_elements_right<T, false>(m_pData, idxDst - 1, idxSrc, m_cap);
367 mem::move_element<T, false>((uint8_t*)ptr, (uint8_t*)&value, 0, 0, 1, 1);
368 }
369
370 ++m_cnt;
371
372 return iterator(&data()[idxSrc]);
373 }

406 iterator erase(iterator pos) noexcept {
407 GAIA_ASSERT(pos >= data());
408 GAIA_ASSERT(empty() || (pos < iterator(data() + size())));
409
410 if (empty())
411 return end();
412
413 const auto idxSrc = (size_type)core::distance(begin(), pos);
414 const auto idxDst = (size_type)core::distance(begin(), end()) - 1;
415
416 mem::shift_elements_left<T, false>(m_pData, idxDst, idxSrc, m_cap);
417 // Destroy if it's the last element
418 auto* ptr = &data()[m_cnt - 1];
419 core::call_dtor(ptr);
420 GAIA_MEM_SANI_POP(value_size, m_pData, m_cap, m_cnt);
421
422 --m_cnt;
423
424 return iterator(&data()[idxSrc]);
425 }

431 iterator erase(iterator first, iterator last) noexcept {
432 GAIA_ASSERT(first >= data())
433 GAIA_ASSERT(empty() || (first < iterator(data() + size())));
434 GAIA_ASSERT(last > first);
435 GAIA_ASSERT(last <= iterator(data() + size()));
436
437 if (empty())
438 return end();
439
440 const auto idxSrc = (size_type)core::distance(begin(), first);
441 const auto idxDst = size();
442 const auto cnt = (size_type)(last - first);
443
444 mem::shift_elements_left_fast<T, false>(m_pData, idxDst, idxSrc, cnt, m_cap);
445 // Destroy if it's the last element
446 auto* ptr = &data()[m_cnt - cnt];
447 core::call_dtor_n(ptr, cnt);
448 GAIA_MEM_SANI_POP_N(value_size, data(), m_cap, m_cnt, cnt);
449
450 m_cnt -= cnt;
451
452 return iterator(&data()[idxSrc]);
453 }

461 void shrink_to_fit() {
462 const auto cap = capacity();
463 const auto cnt = size();
464 if (cap == cnt)
465 return;
466
467 auto* pDataOld = m_pData;
468 m_pData = cnt != 0 ? view_policy::template alloc<Allocator>(cnt) : nullptr;
469 if (cnt != 0) {
470 GAIA_MEM_SANI_ADD_BLOCK(value_size, m_pData, cnt, cnt);
471 mem::move_ctor_elements<T>(m_pData, pDataOld, cnt);
472 }
473 view_policy::template free<Allocator>(pDataOld, cap, cnt);
474 m_cap = cnt;
