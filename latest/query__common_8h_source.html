1433 GAIA_NODISCARD QueryLookupHash::Type hash_lookup_key_payload() const {
1434 QueryLookupHash::Type payloadHash = 0;
1435 const bool useTermAccess = uses_term_access_identity();
1436
1437 // Ids & ops
1438 {
1439 QueryLookupHash::Type hash = 0;
1440
1441 for (const auto& pair: lookup_terms_view()) {
1442 const auto op = (QueryLookupHash::Type)pair.op;
1443 const auto access = !useTermAccess ? op
1444 : (op | ((QueryLookupHash::Type)pair.access << 8) |
1445 ((QueryLookupHash::Type)pair.fieldIndex << 16));
1446 hash = core::hash_combine(hash, access);
1447 hash = core::hash_combine(hash, (QueryLookupHash::Type)pair.id.value());
1448 hash = core::hash_combine(hash, (QueryLookupHash::Type)pair.src.value());
1449 hash = core::hash_combine(hash, (QueryLookupHash::Type)pair.entTrav.value());
1450 hash = core::hash_combine(hash, (QueryLookupHash::Type)(uint8_t)pair.travKind);
1451 hash = core::hash_combine(hash, (QueryLookupHash::Type)pair.travDepth);
1452 hash = core::hash_combine(hash, (QueryLookupHash::Type)(uint8_t)pair.matchKind);
1453 }
1454 hash = core::hash_combine(hash, (QueryLookupHash::Type)idsCnt);
1455 hash = core::hash_combine(hash, (QueryLookupHash::Type)(!useTermAccess ? readWriteMask : 0));
1456 hash = core::hash_combine(hash, (QueryLookupHash::Type)cacheSrcTrav);
1457
1458 const bool matchPrefab = (flags & QueryFlags::MatchPrefab) != 0;
1459 hash = core::hash_combine(hash, (QueryLookupHash::Type)matchPrefab);
1460
1461 payloadHash = hash;
1462 }
1463
1464 // Filters
1465 {
1466 QueryLookupHash::Type hash = 0;
1467
1468 for (const auto entity: changed_lookup_view())
1469 hash = core::hash_combine(hash, (QueryLookupHash::Type)entity.value());
1470 hash = core::hash_combine(hash, (QueryLookupHash::Type)changedCnt);
1471
1472 payloadHash = core::hash_combine(payloadHash, hash);
1473 }
1474
1475 // Explicit grouping dependencies
1476 {
1477 QueryLookupHash::Type hash = 0;
1478
1479 for (const auto entity: group_deps_lookup_view())
1480 hash = core::hash_combine(hash, (QueryLookupHash::Type)entity.value());
1481 hash = core::hash_combine(hash, (QueryLookupHash::Type)groupDepCnt);
1482
1483 payloadHash = core::hash_combine(payloadHash, hash);
1484 }
1485
1486 return payloadHash;
1487 }

1990 inline void sort(QueryCtx& ctx) {
1991 const uint32_t idsCnt = ctx.data.idsCnt;
1992 const uint32_t changedCnt = ctx.data.changedCnt;
1993
1994 auto& ctxData = ctx.data;
1995 // Canonicalize degenerate OR queries: a single OR term has AND semantics.
1996 // Rewriting it here keeps ordering/hash behavior identical to an explicit ALL term.
1997 if (idsCnt > 0) {
1998 uint32_t orCnt = 0;
1999 uint32_t orIdx = BadIndex;
2000 GAIA_FOR(idsCnt) {
2001 if (ctxData.terms[i].op != QueryOpKind::Or)
2002 continue;
2003 ++orCnt;
2004 orIdx = i;
2005 if (orCnt > 1)
2006 break;
2007 }
2008
2009 if (orCnt == 1)
2010 ctxData.terms[orIdx].op = QueryOpKind::All;
2011 }
2012
2013 // Sort data. Necessary for correct hash calculation.
2014 // Without sorting query.all<XXX, YYY> would be different than query.all<YYY, XXX>.
2015 // Also makes sure data is in optimal order for query processing.
2016 core::sort(
2017 ctxData.terms.data(), ctxData.terms.data() + ctxData.idsCnt, query_sort_cond{}, //
2018 [&](uint32_t left, uint32_t right) {
2019 core::swap(ctxData.ids[left], ctxData.ids[right]);
2020 core::swap(ctxData.terms[left], ctxData.terms[right]);
2021
2022 // Make sure masks remains correct after sorting
2023 core::swap_bits(ctxData.readWriteMask, left, right);
2024 core::swap_bits(ctxData.as_mask_0, left, right);
2025 core::swap_bits(ctxData.as_mask_1, left, right);
2026 });
2027
2028 if (idsCnt > 0) {
2029 uint32_t i = 0;
2030 while (i < idsCnt && ctxData.terms[i].op == QueryOpKind::All)
2031 ++i;
2032 ctxData.firstOr = (uint8_t)i;
2033 while (i < idsCnt && ctxData.terms[i].op == QueryOpKind::Or)
2034 ++i;
2035 ctxData.firstNot = (uint8_t)i;
2036 while (i < idsCnt && ctxData.terms[i].op == QueryOpKind::Not)
2037 ++i;
2038 ctxData.firstAny = (uint8_t)i;
2039 } else
2040 ctxData.firstOr = ctxData.firstNot = ctxData.firstAny = 0;
2041
2042 // Canonicalize filter order. This enables monotonic component lookup in filter matching
2043 // and keeps cache keys stable regardless of changed() call order.
2044 if (changedCnt > 1) {
2045 core::sort(ctxData.changed.data(), ctxData.changed.data() + changedCnt, SortComponentCond{});
2046 }
2047
2048 GAIA_FOR(changedCnt) {
2049 const auto comp = ctxData.changed[i];
2050 uint32_t compIdx = 0;
2051 while (compIdx < idsCnt && ctxData.ids[compIdx] != comp)
2052 ++compIdx;
2053
2054 GAIA_ASSERT(compIdx < idsCnt);
2055 ctxData.changedFields[i] = compIdx < idsCnt ? (uint8_t)compIdx : (uint8_t)0xFF;
2056 }
2057 }
