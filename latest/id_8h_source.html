375 // NOTE: Entity::load() only receives a serializer, not World state.
376 // Therefore the load-time core-id remap currently uses scoped ambient context.
377 // thread_local keeps concurrent world loads on different threads independent.
378 // Tradeoff: this is less explicit than carrying the remap state on the serializer.
379 inline thread_local EntityLoadRemapState g_entityLoadRemapState{};
380
381 struct EntityLoadRemapGuard {
382 EntityLoadRemapState prev;
383
384 EntityLoadRemapGuard(
385 uint32_t savedLastCoreComponentId, uint32_t currLastCoreComponentId, bool remapComponentIds) noexcept:
386 prev(g_entityLoadRemapState) {
387 g_entityLoadRemapState.savedLastCoreComponentId = savedLastCoreComponentId;
388 g_entityLoadRemapState.currLastCoreComponentId = currLastCoreComponentId;
389 g_entityLoadRemapState.remapComponentIds = remapComponentIds;
390 g_entityLoadRemapState.active = true;
391 }
392
393 ~EntityLoadRemapGuard() {
394 g_entityLoadRemapState = prev;
395 }
396
397 EntityLoadRemapGuard(const EntityLoadRemapGuard&) = delete;
398 EntityLoadRemapGuard& operator=(const EntityLoadRemapGuard&) = delete;
399 EntityLoadRemapGuard(EntityLoadRemapGuard&&) = delete;
400 EntityLoadRemapGuard& operator=(EntityLoadRemapGuard&&) = delete;
401 };

403 GAIA_NODISCARD inline uint32_t remap_loaded_entity_id(
404 uint32_t id, uint32_t savedLastCoreComponentId, uint32_t currLastCoreComponentId) noexcept {
405 if (id == Entity::IdMask || //
406 id <= savedLastCoreComponentId || //
407 currLastCoreComponentId <= savedLastCoreComponentId //
408 )
409 return id;
410
411 return id + (currLastCoreComponentId - savedLastCoreComponentId);
412 }
413
414 GAIA_NODISCARD inline uint32_t remap_loaded_entity_id(uint32_t id) noexcept {
415 const auto& state = g_entityLoadRemapState;
416 if (!state.active)
417 return id;
418
419 return remap_loaded_entity_id(id, state.savedLastCoreComponentId, state.currLastCoreComponentId);
420 }
421
422 GAIA_NODISCARD inline Entity
423 remap_loaded_entity(Entity entity, uint32_t savedLastCoreComponentId, uint32_t currLastCoreComponentId) noexcept {
424 if (entity == EntityBad)
425 return entity;
426
427 if (!entity.pair()) {
428 return Entity(
429 (EntityId)remap_loaded_entity_id(entity.id(), savedLastCoreComponentId, currLastCoreComponentId),
430 entity.gen(), entity.entity(), false);
431 }
432
433 return Entity(
434 (EntityId)remap_loaded_entity_id(entity.id(), savedLastCoreComponentId, currLastCoreComponentId),
435 (IdentifierData)remap_loaded_entity_id(entity.gen(), savedLastCoreComponentId, currLastCoreComponentId),
436 false, true);
437 }
438
439 GAIA_NODISCARD inline Entity remap_loaded_entity(Entity entity) noexcept {
440 const auto& state = g_entityLoadRemapState;
441 if (!state.active)
442 return entity;
443
444 return remap_loaded_entity(entity, state.savedLastCoreComponentId, state.currLastCoreComponentId);
445 }

709 // Runtime primitive type entities. Supported entity ids are aligned with ser::serialization_type_id:

713 GAIA_NODISCARD inline bool is_runtime_primitive_serialization_type_id(uint32_t typeId) noexcept {
714 return typeId >= (uint32_t)ser::serialization_type_id::s8 && typeId <= (uint32_t)ser::serialization_type_id::f64;
715 }
716
717 GAIA_NODISCARD inline Entity runtime_primitive_type_entity(ser::serialization_type_id type) noexcept {
718 const auto typeId = (uint32_t)type;
719 if (!is_runtime_primitive_serialization_type_id(typeId))
720 return EntityBad;
721 return Entity((EntityId)(RuntimePrimitiveTypeBaseId + typeId), 0, false, false);
722 }
723
724 GAIA_NODISCARD inline bool
725 runtime_primitive_serialization_type(Entity type, ser::serialization_type_id& out) noexcept {
726 if (type.pair())
727 return false;
728
729 const auto typeId = type.id() - RuntimePrimitiveTypeBaseId;
730 if (!is_runtime_primitive_serialization_type_id(typeId))
731 return false;
732
733 out = (ser::serialization_type_id)typeId;
734 return true;
735 }
